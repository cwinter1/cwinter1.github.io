#  CONCEPT 1 — ROUTERS, PYDANTIC MODELS, AND AUTOMATIC DOCS
# ══════════════════════════════════════════════════════════════
# FastAPI uses Python type hints to:
#   - Validate incoming request bodies automatically
#   - Serialize outgoing responses
#   - Generate OpenAPI docs at /docs (no extra code)
#
# APIRouter groups related routes under a shared prefix.
# Attach a router with app.include_router(router).
#
# Pydantic models define the shape of request and response bodies.
# Request model: used in POST/PUT body
# Response model: declared in the route decorator to filter output fields
#
# HTTP status codes:
#   200 OK            -- successful GET
#   201 Created       -- successful POST that created a resource
#   404 Not Found     -- resource does not exist
#   422 Unprocessable -- Pydantic validation failed
#
# EXAMPLE ──────────────────────────────────────────────────────

# In-memory store for the example
_employees: dict[int, dict] = {
    1: {'emp_id': 1, 'name': 'Alice Cohen',   'dept': 'Engineering', 'salary': 95000},
    2: {'emp_id': 2, 'name': 'Bob Levy',       'dept': 'Sales',       'salary': 62000},
    3: {'emp_id': 3, 'name': 'Carol Shapiro',  'dept': 'Finance',     'salary': 78000},
}


class EmployeeCreate(BaseModel):
    name: str = Field(..., min_length=2)
    dept: str
    salary: float = Field(..., gt=0)


class EmployeeResponse(BaseModel):
    emp_id: int
    name: str
    dept: str
    salary: float


emp_router = APIRouter(prefix='/employees', tags=['employees'])


@emp_router.get('/{emp_id}', response_model=EmployeeResponse)
def get_employee(emp_id: int):
    if emp_id not in _employees:
        raise HTTPException(status_code=404, detail=f'Employee {emp_id} not found')
    return _employees[emp_id]


@emp_router.post('/', response_model=EmployeeResponse, status_code=201)
def create_employee(body: EmployeeCreate):
    new_id = max(_employees.keys()) + 1
    record = {'emp_id': new_id, **body.model_dump()}
    _employees[new_id] = record
    return record


app1 = FastAPI(title='Employee Service')
app1.include_router(emp_router)

#  CONCEPT 2 — AUTH HEADER DEPENDENCY AND ERROR HANDLERS
# ══════════════════════════════════════════════════════════════
# FastAPI Depends() lets you extract and validate request components
# before the route function runs. Use it for:
#   - Auth token validation
#   - Database session injection
#   - Rate limiting
#
# Header(...) reads a specific HTTP header.
# FastAPI auto-converts Authorization -> authorization (lowercase).
#
# Global exception handlers override the default response shape.
# RequestValidationError fires when Pydantic rejects a request body.
# Without a custom handler, FastAPI returns a verbose 422 JSON.
# With a custom handler, you return a consistent {error, details} shape.
#
# Auth pattern (Bearer token):
#   Authorization: Bearer <token>
#   - Check header exists
#   - Check it starts with 'Bearer '
#   - Check token is in the allowed set
#
# EXAMPLE ──────────────────────────────────────────────────────

VALID_TOKENS = {'secret-token-alpha', 'secret-token-beta'}


def require_token(authorization: str = Header(...)):
    if not authorization.startswith('Bearer '):
        raise HTTPException(status_code=401, detail='Missing Bearer prefix')
    token = authorization[len('Bearer '):]
    if token not in VALID_TOKENS:
        raise HTTPException(status_code=401, detail='Invalid token')
    return token


app2 = FastAPI(title='Auth Demo')


@app2.exception_handler(RequestValidationError)
async def validation_handler(request: Request, exc: RequestValidationError):
    return JSONResponse(
        status_code=422,
        content={
            'error': 'validation_failed',
            'details': exc.errors(),
        },
    )


class ReportRequest(BaseModel):
    report_name: str
    period: str


@app2.post('/reports/', dependencies=[Depends(require_token)])
def create_report(body: ReportRequest):
    return {'report': body.report_name, 'period': body.period, 'status': 'queued'}

# ══════════════════════════════════════════════════════════════
#  CONCEPT 3 — TestClient INTEGRATION TESTS AS A SUITE
# ══════════════════════════════════════════════════════════════
# TestClient is the same as making real HTTP requests but in-process.
# This means you can write a full integration test suite as plain Python:
#   - No mocking of the app layer
#   - Tests exercise real routing, validation, and error handling
#   - Output reads like a test runner: PASS / FAIL per named test
#
# Pattern:
#   1. Build the app with all routes
#   2. Define assert_eq(label, actual, expected) to print PASS/FAIL
#   3. Name each test clearly: test_create_pipeline, test_delete_missing, etc.
#   4. Run all tests inline -- failures do not stop the suite
#
# Why this matters for FDE work:
#   You ship a service at a customer site in a few hours.
#   You need confidence it works before handing over.
#   A 10-line test suite that runs in <1 second gives that confidence.
#
# EXAMPLE ──────────────────────────────────────────────────────

_pipelines: dict[str, dict] = {}


class PipelineCreate(BaseModel):
    name: str
    source: str
    env: str = Field(..., pattern='^(prod|staging|dev)$')


class PipelineRecord(BaseModel):
    run_id: str
    name: str
    source: str
    env: str


app4 = FastAPI(title='Pipeline Service')


@app4.get('/pipelines/')
def list_pipelines():
    return {'pipelines': list(_pipelines.values()), 'count': len(_pipelines)}


@app4.post('/pipelines/', response_model=PipelineRecord, status_code=201)
def create_pipeline(body: PipelineCreate):
    run_id = str(uuid.uuid4())[:8]
    record = {'run_id': run_id, **body.model_dump()}
    _pipelines[run_id] = record
    return record


@app4.delete('/pipelines/{run_id}')
def delete_pipeline(run_id: str):
    if run_id not in _pipelines:
        raise HTTPException(status_code=404, detail=f'Run {run_id} not found')
    _pipelines.pop(run_id)
    return {'deleted': run_id}


client4 = TestClient(app4, raise_server_exceptions=False)

results = {'pass': 0, 'fail': 0}


def assert_eq(label, actual, expected):
    if actual == expected:
        print(f'  PASS  {label}')
        results['pass'] += 1
    else:
        print(f'  FAIL  {label}  (got {actual!r}, expected {expected!r})')
        results['fail'] += 1


def test_list_empty():
    r = client4.get('/pipelines/')
    assert_eq('list empty', r.json()['count'], 0)


def test_create_pipeline():
    r = client4.post('/pipelines/', json={'name': 'sales_etl', 'source': 'raw.sales', 'env': 'prod'})
    assert_eq('create status', r.status_code, 201)
    assert_eq('create name', r.json()['name'], 'sales_etl')


def test_invalid_env():
    r = client4.post('/pipelines/', json={'name': 'x', 'source': 'y', 'env': 'live'})
    assert_eq('invalid env', r.status_code, 422)


def test_list_after_create():
    r = client4.get('/pipelines/')
    assert_eq('list count after create', r.json()['count'], 1)


def test_delete_pipeline():
    run_id = list(_pipelines.keys())[0]
    r = client4.delete(f'/pipelines/{run_id}')
    assert_eq('delete status', r.status_code, 200)
    assert_eq('list after delete', client4.get('/pipelines/').json()['count'], 0)


def test_delete_missing():
    r = client4.delete('/pipelines/no-such-id')
    assert_eq('delete missing', r.status_code, 404)

# ══════════════════════════════════════════════════════════════
#  CONCEPT 1 — MULTI-STAGE DOCKERFILE FOR A FastAPI SERVICE
# ══════════════════════════════════════════════════════════════
# A single-stage image installs build tools and source code in one layer.
# The final image is large and carries unnecessary build dependencies.
#
# Multi-stage builds solve this:
#   Stage 1 (builder): install dependencies into an isolated directory
#   Stage 2 (runtime): copy only the installed packages into a slim base
#
# Result: the runtime image has no pip, no gcc, no build headers.
# Typical size reduction: 800MB -> 120MB for a Python service.
#
# Security practices baked into the runtime stage:
#   - python:3.11-slim base (no unnecessary OS packages)
#   - Non-root user (appuser) -- containers run as root by default
#   - HEALTHCHECK instruction -- lets the orchestrator restart failed containers
#   - No secrets in ENV -- secrets are injected at runtime via cloud provider
#
# HEALTHCHECK instruction:
#   HEALTHCHECK --interval=30s --timeout=5s --retries=3 \
#     CMD curl -f http://localhost:8000/health || exit 1
#   If /health returns non-200, Docker marks the container unhealthy.
#   Cloud Run and K8s use this signal to replace the pod.
#
# EXAMPLE ──────────────────────────────────────────────────────

MULTISTAGE_DOCKERFILE = '''\
# ── Stage 1: install dependencies ──────────────────────────────
FROM python:3.11 AS builder

WORKDIR /build
COPY requirements.txt .
RUN pip install --target /install/packages --no-cache-dir -r requirements.txt

# ── Stage 2: slim runtime image ────────────────────────────────
FROM python:3.11-slim AS runtime

# Copy only installed packages from builder -- no pip, no gcc
COPY --from=builder /install/packages /usr/local/lib/python3.11/site-packages

WORKDIR /app
COPY main.py .

# Non-root user for security
RUN adduser --disabled-password --gecos "" appuser
USER appuser

EXPOSE 8000

HEALTHCHECK --interval=30s --timeout=5s --retries=3 \\
    CMD curl -f http://localhost:8000/health || exit 1

CMD ["uvicorn", "main:app", "--host", "0.0.0.0", "--port", "8000"]
'''

REQUIREMENTS_TXT = 'fastapi\nuvicorn[standard]\npydantic\n'

MAIN_PY = '''\
from fastapi import FastAPI

app = FastAPI()

@app.get('/health')
def health():
    return {'status': 'ok'}

@app.get('/ready')
def ready():
    return {'status': 'ready'}
'''

# Write files to temp dir and show what the build commands look like
tmpdir = tempfile.mkdtemp()
for fname, content in [('Dockerfile', MULTISTAGE_DOCKERFILE),
                        ('requirements.txt', REQUIREMENTS_TXT),
                        ('main.py', MAIN_PY)]:
    with open(os.path.join(tmpdir, fname), 'w') as f:
        f.write(content)

print('Files written to:', tmpdir)
print()
print('Dockerfile contents:')
print(MULTISTAGE_DOCKERFILE)
print()

# ══════════════════════════════════════════════════════════════
#  CONCEPT 2 — BUILD, TAG, PUSH, AND THE CONTAINER REGISTRY
# ══════════════════════════════════════════════════════════════
# A container registry stores versioned images. Common registries:
#   - Google Artifact Registry (gcr.io or region-docker.pkg.dev)
#   - AWS ECR (account_id.dkr.ecr.region.amazonaws.com)
#   - Docker Hub (hub.docker.com)
#
# Tagging strategy for production:
#   1. Semantic version tag:   my-service:1.2.3
#   2. latest tag:             my-service:latest  (convenience, not for prod deploy)
#   3. Git commit SHA tag:     my-service:abc1234  (pinned, fully reproducible)
#
# Never deploy latest to production without also pushing a version or SHA tag.
# If a rollback is needed, you need an immutable tag to roll back to.
#
# Workflow:
#   docker build -t image:version .
#   docker tag image:version registry/project/image:version
#   docker tag image:version registry/project/image:latest
#   gcloud auth configure-docker REGION-docker.pkg.dev
#   docker push registry/project/image:version
#   docker push registry/project/image:latest
#
# EXAMPLE ──────────────────────────────────────────────────────

def build_image_tags(project: str, repo: str, image_name: str,
                     version: str, commit_sha: str) -> dict:
    """Returns a dict of tag aliases for a versioned image."""
    registry = f'europe-west1-docker.pkg.dev/{project}/{repo}'
    short_sha = commit_sha[:7]
    return {
        'version': f'{registry}/{image_name}:{version}',
        'latest':  f'{registry}/{image_name}:latest',
        'commit':  f'{registry}/{image_name}:{short_sha}',
    }


def simulate_push_workflow(project: str, repo: str, image_name: str,
                           version: str, commit_sha: str) -> None:
    tags = build_image_tags(project, repo, image_name, version, commit_sha)
    local = f'{image_name}:{version}'
    region = 'europe-west1'

    steps = [
        (f'docker build -t {local} .', '  [+] Building 12.7s\n  => exporting to image'),
        (f'docker tag {local} {tags["version"]}', ''),
        (f'docker tag {local} {tags["latest"]}', ''),
        (f'gcloud auth configure-docker {region}-docker.pkg.dev',
         '  Adding credentials for: europe-west1-docker.pkg.dev'),
        (f'docker push {tags["version"]}',
         f'  The push refers to repository [{tags["version"].rsplit(":", 1)[0]}]\n'
         f'  {version}: digest: sha256:a1b2c3... size: 1847'),
        (f'docker push {tags["latest"]}',
         f'  latest: digest: sha256:a1b2c3... size: 1847'),
    ]

    print(f'--- Push workflow for {image_name} v{version} ---')
    for cmd, output in steps:
        print(f'$ {cmd}')
        if output:
            print(output)
    print()
    print('Tags pushed:')
    for alias, tag in tags.items():
        print(f'  {alias:8s}  {tag}')


simulate_push_workflow(
    project='my-gcp-project',
    repo='services',
    image_name='sales-api',
    version='2.1.0',
    commit_sha='abc1234def5678',
)

#  CONCEPT 3 — DEPLOYING TO CLOUD RUN AND RAILWAY
# ══════════════════════════════════════════════════════════════
# Cloud Run (Google) -- serverless containers:
#   - You push an image; Cloud Run pulls it and runs it.
#   - Scales to zero by default (cost savings), or set --min-instances=1.
#   - Traffic routed by revision -- can split traffic for canary deploys.
#   - IAM controls who can invoke the service.
#   - Good for: production APIs, background workers, event-driven pipelines.
#
# Railway -- developer-focused PaaS:
#   - Detects framework automatically (FastAPI, Express, etc.).
#   - No infra config required -- just push code or an image.
#   - PR preview environments built-in.
#   - Free tier available; good for prototypes and demos.
#   - Good for: customer demos, side projects, shipping an MVP fast.
#
# For FDE work, Railway is often the right call for a same-day customer demo.
# Move to Cloud Run when compliance, SLAs, or cost governance kick in.

def _service_url_hash(service_name: str, region: str) -> str:
    """Deterministic hash-based URL for simulated output."""
    h = hashlib.md5(f'{service_name}-{region}'.encode()).hexdigest()[:8]
    return f'https://{service_name}-{h}-uc.a.run.app'


def simulate_cloud_run_deploy(service_name: str, image: str, region: str,
                               port: int, memory: str, env_vars: dict) -> str:
    env_flags = ' '.join(f'{k}=${k}' for k in env_vars)
    cmd = (
        f'gcloud run deploy {service_name}\\\n'
        f'  --image={image}\\\n'
        f'  --platform=managed\\\n'
        f'  --region={region}\\\n'
        f'  --port={port}\\\n'
        f'  --memory={memory}\\\n'
        f'  --min-instances=1\\\n'
        f'  --set-env-vars={env_flags}'
    )
    url = _service_url_hash(service_name, region)
    print(f'--- Cloud Run: {service_name} ---')
    print(f'$ {cmd}')
    print(f'  Deploying container to Cloud Run service [{service_name}] in [{region}]')
    print(f'  Service [{service_name}] revision [{service_name}-00001-abc] has been deployed')
    print(f'  Service URL: {url}')
    print()
    return url


def simulate_railway_deploy(service_name: str, env_vars: dict) -> str:
    h = hashlib.md5(service_name.encode()).hexdigest()[:6]
    url = f'https://{service_name}-production-{h}.up.railway.app'
    var_set = ' '.join(f'{k}=${k}' for k in env_vars)
    print(f'--- Railway: {service_name} ---')
    print(f'$ railway init --name {service_name}')
    print(f'  Created project {service_name}')
    print(f'$ railway up')
    print(f'  Deploying {service_name}...')
    print(f'  Build: SUCCESS')
    print(f'$ railway vars set {var_set}')
    print(f'  Set {len(env_vars)} variable(s)')
    print(f'$ railway domain')
    print(f'  {url}')
    print()
    return url

# ══════════════════════════════════════════════════════════════
#  CONCEPT 4 — SECURITY GATE: DEPLOYMENT SECURITY GATE (CIA: C+I+A)
# ══════════════════════════════════════════════════════════════
# Analogy: the car safety inspection before road registration — brakes,
# lights, and seatbelts are mandatory; no certificate means no road.
#
# CIA focus: All three — the container and API must pass an OWASP-aligned
# security checklist before any deployment.
#
# Checks:
#   1. Dockerfile must have a USER directive and it must not be 'root'
#   2. Debug routes (/docs, /redoc) must not be unauthenticated
#   3. Routes returning > 15 fields risk mass assignment (OWASP API #3)
#
# EXAMPLE ──────────────────────────────────────────────────────────
from dataclasses import dataclass, field


class RootContainerError(Exception):
    pass


class ExposedDebugRouteError(Exception):
    pass


class MassAssignmentRisk(Exception):
    pass


@dataclass
class DeployReport:
    passed: bool
    violations: list = field(default_factory=list)


def deployment_security_gate(
    dockerfile_lines: list,
    api_routes: list,
) -> DeployReport:
    user_lines = [l for l in dockerfile_lines if l.strip().upper().startswith("USER")]
    if not user_lines or any("root" in l.lower() for l in user_lines):
        raise RootContainerError(
            "Container runs as root -- add 'USER appuser' to Dockerfile"
        )
    for route in api_routes:
        if route["path"] in ("/docs", "/redoc") and not route.get("has_auth", True):
            raise ExposedDebugRouteError(
                f"Debug route '{route['path']}' is unauthenticated"
            )
        if len(route.get("returns_fields", [])) > 15:
            raise MassAssignmentRisk(
                f"Route '{route['path']}' returns {len(route['returns_fields'])} fields"
            )
    return DeployReport(passed=True)


# Secure deployment — passes
secure_dockerfile = [
    "FROM python:3.11-slim",
    "WORKDIR /app",
    "COPY . .",
    "USER appuser",
    "HEALTHCHECK CMD curl -f http://localhost:8000/health || exit 1",
]
secure_routes = [
    {"path": "/api/orders", "has_auth": True, "returns_fields": ["id", "status"]},
]
report = deployment_security_gate(secure_dockerfile, secure_routes)
print(f"Secure deployment: passed={report.passed}")

# Root container — blocked
try:
    deployment_security_gate(["FROM python:3.11-slim", "USER root"], [])
except RootContainerError as e:
    print(f"RootContainerError: {e}")
