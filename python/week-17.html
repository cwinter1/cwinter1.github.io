# ══════════════════════════════════════════════════════════════
#  CONCEPT 1 — WHAT IDEMPOTENCY MEANS
# ══════════════════════════════════════════════════════════════
# An operation is idempotent if running it N times produces the same result as running it once.
#
#   Idempotent:     PUT /users/42 with {name: 'Alice'}  -- always results in Alice at id=42
#   NOT idempotent: POST /users    with {name: 'Alice'}  -- creates a new user every time
#
# In data pipelines, idempotency matters when:
#   - A task is retried after failure (Airflow retry)
#   - A backfill re-runs historical data
#   - A bug is fixed and the pipeline is re-run for the same date
#
# Non-idempotent loads cause:
#   - Duplicate rows in tables
#   - Inflated aggregates (revenue counted twice)
#   - Dashboard trust issues
#
# Three strategies to make a load idempotent:
#
#   1. WRITE_TRUNCATE (drop + reload)
#        DROP TABLE IF EXISTS stg_sales; CREATE TABLE stg_sales AS SELECT ...;
#        Safe: full replace. Only for tables owned by this pipeline.
#
#   2. UPSERT / MERGE
#        UPDATE rows that exist, INSERT rows that don't.
#        Preserves existing data. Best for shared tables.
#
#   3. Partition-level REPLACE
#        DELETE FROM events WHERE date = '2024-01-15'; INSERT ...
#        BigQuery equivalent: WRITE_TRUNCATE on a single partition.
#
# EXAMPLE ──────────────────────────────────────────────────────

# Show why a plain INSERT is NOT idempotent
db.execute("""
    CREATE TABLE orders_naive (
        order_id  INTEGER,
        product   VARCHAR,
        amount    DOUBLE
    )
""")

    """Non-idempotent: inserts every time it is called."""

# ══════════════════════════════════════════════════════════════
#  CONCEPT 2 — MERGE/UPSERT PATTERN
# ══════════════════════════════════════════════════════════════
# UPSERT = UPDATE existing rows + INSERT new rows.
# In DuckDB (and most SQL databases) you can do this with:
#
#   INSERT OR REPLACE INTO target SELECT ... FROM source;
#
# Or with an explicit MERGE statement (Postgres, BigQuery, Snowflake):
#   MERGE INTO target AS t
#   USING source AS s ON t.id = s.id
#   WHEN MATCHED THEN UPDATE SET t.amount = s.amount
#   WHEN NOT MATCHED THEN INSERT VALUES (s.id, s.amount);
#
# DuckDB INSERT OR REPLACE requires a PRIMARY KEY constraint.
# Without a primary key, use DELETE + INSERT:
#   DELETE FROM target WHERE key_col IN (SELECT key_col FROM new_batch);
#   INSERT INTO target SELECT * FROM new_batch;
#
# BigQuery equivalent:
#   bq_client.load_table_from_dataframe(df, 'dataset.table',
#       job_config=bigquery.LoadJobConfig(
#           write_disposition='WRITE_TRUNCATE',   # full replace
#       ))
#
# EXAMPLE ──────────────────────────────────────────────────────

# Idempotent load using WRITE_TRUNCATE (drop + recreate)
def load_truncate(df, table):
    """Idempotent: completely replaces the table every run."""
    db.execute(f'CREATE OR REPLACE TABLE {table} AS SELECT * FROM df')
    return db.execute(f'SELECT COUNT(*) FROM {table}').fetchone()[0]

print('--- WRITE_TRUNCATE (idempotent) ---')
print(f'After run 1: {load_truncate(batch, "orders_safe")} rows')
print(f'After run 2: {load_truncate(batch, "orders_safe")} rows')   # same
print(f'After run 3: {load_truncate(batch, "orders_safe")} rows')   # same
print()

# Idempotent UPSERT using DELETE + INSERT
db.execute("""
    CREATE TABLE orders_upsert (
        order_id  INTEGER PRIMARY KEY,
        product   VARCHAR,
        amount    DOUBLE
    )
""")

def upsert_load(df, table, key_col):
    """
    Idempotent load: delete existing keys then insert.
    Run it 10 times, get the same table.
    """
    keys = ', '.join(str(k) for k in df[key_col].tolist())
    db.execute(f'DELETE FROM {table} WHERE {key_col} IN ({keys})')
    db.execute(f'INSERT INTO {table} SELECT * FROM df')
    return db.execute(f'SELECT COUNT(*) FROM {table}').fetchone()[0]

# ══════════════════════════════════════════════════════════════
#  CONCEPT 3 — IDEMPOTENT LOAD FUNCTIONS
# ══════════════════════════════════════════════════════════════
# A production-grade idempotent load function handles:
#   - Table creation on first run (if not exists)
#   - WRITE_TRUNCATE for full-replace tables
#   - UPSERT for incrementally-updated tables
#   - Deduplication WITHIN the incoming batch (in case source sends duplicates)
#
# Deduplication within the batch:
#   df = df.drop_duplicates(subset='order_id', keep='last')
#
# Partition-level idempotency (BigQuery / partitioned tables):
#   DELETE FROM events WHERE DATE(event_ts) = '2024-01-15';
#   INSERT INTO events SELECT * FROM new_events;
#   -- Same day re-runs are safe: delete clears the partition first.
#
# Design rule: every pipeline should answer YES to:
#   'If I run this twice with the same input, is the output identical?'
#
# EXAMPLE ──────────────────────────────────────────────────────

def idempotent_load(df, table, key_col, mode='upsert', create_if_missing=True):
    """
    General-purpose idempotent load.

    mode='truncate'  -- drop and recreate (full replace)
    mode='upsert'    -- delete matching keys, then insert

    Automatically deduplicates the incoming df on key_col.
    Creates the table from df schema on first run if it does not exist.
    """
    # Step 1: dedup the incoming batch
    original_len = len(df)
    df = df.drop_duplicates(subset=key_col, keep='last')
    if len(df) < original_len:
        print(f'  [dedup] dropped {original_len - len(df)} duplicate rows in incoming batch')

    # Step 2: ensure table exists
    tables = [r[0] for r in db.execute('SHOW TABLES').fetchall()]
    if table not in tables and create_if_missing:
        db.execute(f'CREATE TABLE {table} AS SELECT * FROM df LIMIT 0')
        print(f'  [create] table {table} created from schema')

    # Step 3: apply the selected strategy
    if mode == 'truncate':
        db.execute(f'CREATE OR REPLACE TABLE {table} AS SELECT * FROM df')
    elif mode == 'upsert':
        keys = ', '.join(str(k) for k in df[key_col].tolist())
        db.execute(f'DELETE FROM {table} WHERE {key_col} IN ({keys})')
        db.execute(f'INSERT INTO {table} SELECT * FROM df')

    n = db.execute(f'SELECT COUNT(*) FROM {table}').fetchone()[0]
    return n

# ══════════════════════════════════════════════════════════════
#  CONCEPT 1 — MEMORY PROFILING
# ══════════════════════════════════════════════════════════════
# df.memory_usage(deep=True) returns bytes used per column.
# 'deep=True' counts the actual string content of object columns
# (without it, object columns show 8 bytes per row regardless of string length).
#
# Dtype memory costs (per value):
#   object (Python str)   ~50+ bytes per value (pointer + string object)
#   float64               8 bytes
#   int64                 8 bytes
#   int32                 4 bytes
#   float32               4 bytes
#   bool                  1 byte
#   category              varies — stores an integer index + a lookup table
#                         Huge savings when cardinality is low (< 50 unique values)
#
# Rule of thumb:
#   object column with < 50 unique values   -> convert to category
#   int64 column max value < 2 billion      -> downcast to int32
#   float64 column                          -> downcast to float32 if precision allows
#
# EXAMPLE ──────────────────────────────────────────────────────

def memory_report(df, label=''):
    """Print a per-column memory usage report."""
    mem = df.memory_usage(deep=True)
    total_kb = mem.sum() / 1024
    print(f'{label}  total: {total_kb:.1f} KB')
    for col in df.columns:
        col_kb = mem[col] / 1024
        print(f'  {col:<20}  {str(df[col].dtype):<12}  {col_kb:>7.2f} KB  '
              f'unique: {df[col].nunique()}')
    print()

# ══════════════════════════════════════════════════════════════
#  CONCEPT 3 — EFFICIENT DTYPES: AUTO-DOWNCAST
# ══════════════════════════════════════════════════════════════
# Manually converting every column is tedious.
# Write a function that inspects each column and applies the best dtype:
#
#   int64 column:    if max <= 2^31 - 1 (2,147,483,647)  -> downcast to int32
#   float64 column:  downcast to float32 if precision loss is acceptable
#   object column:   if unique count / total rows < threshold -> convert to category
#
# float32 vs float64:
#   float32 has ~7 significant digits. For currency values stored to 2 decimal places,
#   float32 is usually sufficient. For scientific computing, keep float64.
#
# category threshold:
#   A common rule: convert to category if cardinality < 50 unique values
#   OR if unique_count / len(df) < 0.5 (less than 50% unique).
#
# EXAMPLE ──────────────────────────────────────────────────────

def optimize_dtypes(df, category_max_unique=50):
    """
    Auto-downcast:
      int64   -> int32   (if values fit)
      float64 -> float32
      object  -> category (if unique count <= category_max_unique)
    Returns a new DataFrame with optimized dtypes.
    """
    df = df.copy()
    for col in df.columns:
        col_dtype = df[col].dtype

        if col_dtype == 'int64':
            if df[col].max() <= np.iinfo('int32').max:
                df[col] = df[col].astype('int32')

        elif col_dtype == 'float64':
            df[col] = df[col].astype('float32')

        elif col_dtype == 'object':
            if df[col].nunique() <= category_max_unique:
                df[col] = df[col].astype('category')

    return df

#

#   < 1 GB             pandas      simple, well-known, rich ecosystem
#   1 GB – ~50 GB      Polars      10-100x faster than pandas, still single machine

#

#   - Runs on a cluster — driver node sends tasks to worker nodes
#   - DataFrames are distributed — each partition lives on a different machine

#  CONCEPT 1 — PIPELINE HEALTH METRICS
# ══════════════════════════════════════════════════════════════
# Every pipeline run should emit a structured metrics object.
# Key metrics to capture:
#
#   rows_in       how many rows entered the pipeline
#   rows_out      how many rows were successfully processed and written
#   rows_dropped  rows rejected by validation
#   errors        count of exceptions caught during the run
#   start_time    UTC timestamp when the run began
#   end_time      UTC timestamp when the run ended
#   duration_s    wall-clock seconds from start to end
#
# Derived health signals:
#   pass_rate     rows_out / rows_in  (expect >= 0.95 for healthy data)
#   drop_rate     rows_dropped / rows_in
#
# Why capture this:
#   - A pipeline that silently drops 20% of rows is a business incident
#   - Duration spikes often indicate upstream data volume anomalies
#   - Error counts should be 0 for a healthy run
#
# EXAMPLE ──────────────────────────────────────────────────────

@dataclass
class PipelineMetrics:
    """Captures health metrics for a single pipeline run."""
    pipeline_id : str
    start_time  : str = field(default_factory=lambda: datetime.utcnow().isoformat())
    end_time    : Optional[str] = None
    rows_in     : int = 0
    rows_out    : int = 0
    rows_dropped: int = 0
    errors      : int = 0
    status      : str = 'running'

    def finish(self, status='success'):
        self.end_time = datetime.utcnow().isoformat()
        self.status   = status

    @property
    def duration_s(self):
        if not self.end_time:
            return None
        start = datetime.fromisoformat(self.start_time)
        end   = datetime.fromisoformat(self.end_time)
        return round((end - start).total_seconds(), 3)

    @property
    def pass_rate(self):
        if self.rows_in == 0:
            return 0.0
        return round(self.rows_out / self.rows_in, 4)

    def to_dict(self):
        d = asdict(self)
        d['duration_s'] = self.duration_s
        d['pass_rate']  = self.pass_rate
        return d

#  CONCEPT 2 — STRUCTURED LOGGING FOR PIPELINES
# ══════════════════════════════════════════════════════════════
# Plain print() statements are hard to query and aggregate.
# Structured logging emits JSON lines — one JSON object per log event.
# Log aggregators (Datadog, Splunk, CloudWatch) parse these automatically.
#
# JSON log line format:
#   {"ts": "2024-01-15T09:00:01Z", "level": "INFO",
#    "event": "extract_complete", "rows": 1000, "pipeline": "daily_sales"}
#
# Log levels:
#   DEBUG    developer detail (disable in prod)
#   INFO     normal progress events
#   WARNING  something unexpected but not fatal
#   ERROR    a step failed (pipeline may continue or stop)
#   CRITICAL pipeline is in an unrecoverable state
#
# In production you would use Python's logging module:
#   import logging, json
#   logging.basicConfig(level=logging.INFO)
#   logger = logging.getLogger('pipeline')
#   logger.info(json.dumps({'event': 'extract_complete', 'rows': 1000}))
#
# EXAMPLE ──────────────────────────────────────────────────────

LOG_LEVELS = {'DEBUG': 10, 'INFO': 20, 'WARNING': 30, 'ERROR': 40, 'CRITICAL': 50}
MIN_LOG_LEVEL = 'INFO'

def structured_log(event, level='INFO', **kwargs):
    """
    Emit a JSON log line to stdout.
    In production: pass to logging.getLogger().info(...) instead of print.
    """
    if LOG_LEVELS.get(level, 0) < LOG_LEVELS.get(MIN_LOG_LEVEL, 0):
        return   # suppress below minimum level

    record = {
        'ts':    datetime.utcnow().isoformat() + 'Z',
        'level': level,
        'event': event,
    }
    record.update(kwargs)
    print(json.dumps(record))

#  CONCEPT 3 — ALERTING: THRESHOLD CHECKS
# ══════════════════════════════════════════════════════════════
# After a pipeline run, check the PipelineMetrics object against thresholds.
# Raise an AlertError if any threshold is violated.
#
# Common thresholds:
#   pass_rate < 0.95             -- more than 5% of rows dropped
#   duration_s > 60              -- pipeline took more than 1 minute
#   errors > 0                   -- any exception during the run
#   rows_out < 1                 -- zero rows loaded (silent empty run)
#
# Alert destinations in production:
#   Slack webhook    -- instant notification to a channel
#   PagerDuty API    -- on-call engineer page
#   Email (SMTP)     -- stakeholder notification
#   Airflow UI       -- task marked 'failed' with error message
#
# Pattern: check_thresholds() raises, on_failure_callback() catches and routes.
#
# EXAMPLE ──────────────────────────────────────────────────────

class AlertError(Exception):
    """Raised when a pipeline metric crosses a threshold."""
    pass


def check_thresholds(metrics, min_pass_rate=0.95, max_duration_s=60.0):
    """
    Validate PipelineMetrics against thresholds.
    Returns list of alert messages (empty = healthy).
    Raises AlertError if any threshold is violated.
    """
    alerts = []

    if metrics.pass_rate < min_pass_rate:
        alerts.append(
            f'pass_rate {metrics.pass_rate:.4f} < threshold {min_pass_rate} '
            f'({metrics.rows_dropped} rows dropped)'
        )

    if metrics.duration_s is not None and metrics.duration_s > max_duration_s:
        alerts.append(
            f'duration {metrics.duration_s:.1f}s > threshold {max_duration_s}s'
        )

    if metrics.errors > 0:
        alerts.append(f'{metrics.errors} error(s) during run')

    if metrics.rows_out < 1:
        alerts.append('zero rows loaded -- silent empty run')

    if alerts:
        raise AlertError('\n'.join(alerts))

    return []

# ══════════════════════════════════════════════════════════════
#  CONCEPT 4 — SLOs, ERROR BUDGETS, AND BURN RATE
# ══════════════════════════════════════════════════════════════
#
#  SLI → SLO → Error Budget → Burn Rate → Alert
#
#  SLI (Service Level Indicator)
#  ──────────────────────────────
#  A measurable signal: success_rate, p99_latency_ms, pipeline_freshness_min
#  Example: SLI = successful_runs / total_runs over 30 days
#
#  SLO (Service Level Objective)
#  ──────────────────────────────
#  The target you commit to: SLI >= 99.5%
#  Not 100% — that would require zero deploys and no risk-taking.
#
#  Error Budget
#  ─────────────
#  How much failure you are allowed: 1 - SLO
#  SLO = 99.5%  →  error budget = 0.5%
#  Over 30 days = 43,200 minutes  →  budget = 216 minutes of downtime
#
#  Burn Rate
#  ──────────
#  How fast you are consuming the error budget.
#  burn_rate = actual_error_rate / (1 - SLO)
#  burn_rate = 1.0  → consuming budget at exactly the right pace
#  burn_rate = 10.0 → consuming 10x too fast → will exhaust budget in 3 days
#
#  Multi-window alerts (Google SRE Book)
#  ──────────────────────────────────────
#  Alert if burn_rate > threshold over a short AND long window:
#    Fast burn:  1h window,  burn_rate > 14  (exhausts budget in ~2 days)
#    Slow burn: 6h window,  burn_rate > 5   (exhausts budget in ~6 days)
#  Two-window reduces false positives from short spikes.
#
#  Consequence
#  ───────────
#  Budget >= 0.5 left:  deploy freely
#  Budget < 0.5:        only critical fixes, freeze features
#  Budget = 0:          stop all deployments, incident response only
#
# EXAMPLE ──────────────────────────────────────────────────────
from dataclasses import dataclass
from typing import Optional


@dataclass
class SLO:
    name: str
    sli_description: str
    target: float          # e.g. 0.995 for 99.5%
    window_days: int = 30

    @property
    def error_budget_fraction(self) -> float:
        return 1.0 - self.target

    def error_budget_minutes(self) -> float:
        return self.window_days * 24 * 60 * self.error_budget_fraction

    def burn_rate(self, current_error_rate: float) -> float:
        """How fast the error budget is being consumed relative to allowed rate."""
        if self.error_budget_fraction == 0:
            return float("inf")
        return current_error_rate / self.error_budget_fraction

    def budget_remaining(self, errors_so_far: int, total_events: int) -> float:
        """Fraction of error budget still available (0.0 - 1.0)."""
        if total_events == 0:
            return 1.0
        actual_error_rate = errors_so_far / total_events
        consumed = actual_error_rate / self.error_budget_fraction
        return max(0.0, 1.0 - consumed)

    def alert(self, burn_rate_1h: float, burn_rate_6h: float) -> Optional[str]:
        """Multi-window burn rate alert. Returns alert level or None."""
        if burn_rate_1h > 14 and burn_rate_6h > 14:
            return "PAGE — fast burn: budget exhausted in ~2 days"
        if burn_rate_1h > 5 and burn_rate_6h > 5:
            return "TICKET — slow burn: budget exhausted in ~6 days"
        return None


# Pipeline SLO: 99.5% of runs complete without error

#  CONCEPT 1 — QUERY COST ESTIMATION
# ══════════════════════════════════════════════════════════════
# BigQuery bills by bytes scanned (on-demand pricing: $5 per TB).
# Before running a query you can estimate its cost with a dry-run:
#
#   In real BigQuery:
#     from google.cloud import bigquery
#     job_config = bigquery.QueryJobConfig(dry_run=True, use_query_cache=False)
#     job = client.query(sql, job_config=job_config)
#     print(f'This query will process {job.total_bytes_processed} bytes')
#
# Without a BigQuery connection, you can estimate from table sizes:
#   - Parse the FROM / JOIN clauses to find referenced tables
#   - Look up each table's size in a registry
#   - SELECT *   -> scans 100% of the table
#   - SELECT a,b -> scans only those columns (columnar storage helps)
#   - WHERE on partition col -> scans only matching partitions
#
# This is an approximation. Real dry-run is always more accurate.
# Use this pattern in dev/staging to flag expensive queries early.
#
# EXAMPLE ──────────────────────────────────────────────────────

# Table size registry (bytes)
TABLE_SIZES = {
    'events':          2_000_000_000_000,   # 2 TB
    'orders':            500_000_000_000,   # 500 GB
    'customers':          50_000_000_000,   # 50 GB
    'products':            1_000_000_000,   # 1 GB
    'daily_summary':       5_000_000_000,   # 5 GB (pre-aggregated)
}

PRICE_PER_TB    = 5.0
FREE_TB_PER_MO  = 1.0


def extract_tables(sql):
    """Extract table names from FROM and JOIN clauses (simplified)."""
    sql_upper = sql.upper()
    tables    = []
    # Match: FROM table_name and JOIN table_name
    pattern = r'(?:FROM|JOIN)\s+([\w.]+)'
    for match in re.finditer(pattern, sql_upper):
        raw   = match.group(1)
        name  = raw.split('.')[-1].lower()   # strip dataset prefix
        if name in TABLE_SIZES:
            tables.append(name)
    return list(set(tables))


def estimate_query_cost(sql, selectivity=1.0):
    """
    Estimate BigQuery cost from table size registry.

    selectivity: fraction of table scanned (1.0 = full scan).
    Pass 0.1 if you know a WHERE clause filters to ~10% of the table.
    """
    tables = extract_tables(sql)
    if not tables:
        return {'error': 'no recognised tables found', 'estimated_cost_usd': 0}

    total_bytes  = sum(TABLE_SIZES[t] for t in tables) * selectivity
    total_tb     = total_bytes / (1024 ** 4)
    billable_tb  = max(0, total_tb - FREE_TB_PER_MO)
    cost_usd     = billable_tb * PRICE_PER_TB

    return {
        'tables_referenced': tables,
        'bytes_scanned':     int(total_bytes),
        'tb_scanned':        round(total_tb, 4),
        'estimated_cost_usd': round(cost_usd, 4),
    }

#  CONCEPT 2 — COST GUARDRAILS: REJECT EXPENSIVE QUERIES
# ══════════════════════════════════════════════════════════════
# A cost guardrail checks the estimated cost BEFORE executing the query.
# If the estimate exceeds a threshold, raise an error instead of running.
#
# In BigQuery you can also use:
#   job_config = bigquery.QueryJobConfig(
#       maximum_bytes_billed=10_000_000_000   # 10 GB limit
#   )
#   -- query raises google.api_core.exceptions.Forbidden if over limit
#
# Tiered governance:
#   < $1.00   -- allow automatically
#   $1-$10    -- allow with WARNING logged
#   > $10.00  -- require explicit approval flag or reject
#
# EXAMPLE ──────────────────────────────────────────────────────

class CostLimitError(Exception):
    """Raised when a query exceeds the cost guardrail threshold."""
    pass


def cost_guard(sql, max_cost_usd=5.0, warn_cost_usd=1.0, selectivity=1.0):
    """
    Estimate query cost and enforce tiers:
      < warn_cost_usd  -> allow, no message
      >= warn_cost_usd -> allow, print WARNING
      >= max_cost_usd  -> raise CostLimitError

    Returns the cost estimate dict if allowed.
    """
    est = estimate_query_cost(sql, selectivity=selectivity)
    cost = est['estimated_cost_usd']

    if cost >= max_cost_usd:
        raise CostLimitError(
            f'Query rejected: estimated ${cost:.4f} exceeds limit ${max_cost_usd:.2f}. '
            f'Tables: {est["tables_referenced"]}. '
            f'Add partition filters or select fewer columns.'
        )

    if cost >= warn_cost_usd:
        print(f'  [COST WARNING] ${cost:.4f} for tables {est["tables_referenced"]}')

    return est

 CONCEPT 4 — HIGH-THROUGHPUT ASYNC BATCH PROCESSING
══════════════════════════════════════════════════════════════
══════════════════════════════════════════════════════════════

 By default, Python waits for each LLM call to finish before starting
 the next. For batch jobs (summarise 50 documents, classify 1000 rows)
 this is wasteful — the CPU sits idle while waiting for the API.

 asyncio lets you launch multiple calls concurrently:
   async def call_llm(prompt): ...   define an async function
   await call_llm(prompt)            wait for one call
   asyncio.gather(*tasks)            wait for ALL calls concurrently

 In a notebook, use asyncio.run() or await directly (Jupyter supports it).

 Typical speedup: 5-10x for I/O-bound batch jobs.
 Limit concurrency to avoid rate limits: asyncio.Semaphore(max_concurrent)

EXAMPLE ──────────────────────────────────────────────────────
import asyncio
import time
import random

# Simulate an LLM API call (replace with real openai.AsyncOpenAI in production)
async def simulate_llm_call(prompt, call_id):
    """Simulates a 0.5-1.0s API call latency."""
    await asyncio.sleep(random.uniform(0.5, 1.0))
    return {'id': call_id, 'prompt': prompt[:30], 'response': f'Summary {call_id}'}


prompts = [f'Summarise document {i}: Lorem ipsum content...' for i in range(8)]

# ── Sequential (slow) ─────────────────────────────────────────
async def run_sequential(prompts):
    results = []
    for i, p in enumerate(prompts):
        results.append(await simulate_llm_call(p, i))
    return results

# ── Concurrent with semaphore (fast, rate-limited) ────────────
async def run_concurrent(prompts, max_concurrent=4):
    sem = asyncio.Semaphore(max_concurrent)
    async def bounded_call(prompt, i):
        async with sem:
            return await simulate_llm_call(prompt, i)
    return await asyncio.gather(*[bounded_call(p, i) for i, p in enumerate(prompts)])


t0 = time.perf_counter()
seq_results = asyncio.run(run_sequential(prompts))
seq_time = time.perf_counter() - t0

t0 = time.perf_counter()
con_results = asyncio.run(run_concurrent(prompts, max_concurrent=4))
con_time = time.perf_counter() - t0

print(f'Sequential : {seq_time:.2f}s  ({len(seq_results)} results)')
print(f'Concurrent : {con_time:.2f}s  ({len(con_results)} results)')
print(f'Speedup    : {seq_time / con_time:.1f}x')

 CONCEPT 5 — TOKEN BUDGET & CANARY PROMPT DEPLOYMENT
══════════════════════════════════════════════════════════════
══════════════════════════════════════════════════════════════

 TOKEN BUDGET
 Every LLM call costs tokens. Unchecked, a single bad request
 (a 50-page document sent as context) can cost 10x a normal call.
 Enforce a budget before sending:
   estimated_tokens = len(text.split()) * 1.3  (rough heuristic)
   tiktoken.encode(text)                       (exact, OpenAI models)

 TOKEN BUDGET PATTERN:
   if tokens > MAX_TOKENS: truncate or chunk before sending
   always log token count alongside cost estimate

 CANARY DEPLOYMENT
 Before rolling out a new prompt to all traffic, test it on a small
 fraction (5-10%). Compare quality metrics. Roll back if worse.

   router(user_id) -> 'new_prompt' if hash(user_id) % 100 < canary_pct
                   -> 'old_prompt' otherwise

 Using a hash (not random) ensures the SAME user always gets the
 same prompt version — consistent experience, reproducible results.

EXAMPLE ──────────────────────────────────────────────────────

# ── Token budget ──────────────────────────────────────────────
def estimate_tokens(text):
    """Rough estimate: ~1.3 tokens per word (OpenAI rule of thumb)."""

def enforce_token_budget(text, max_tokens=2000, strategy='truncate'):
    """
    Returns (text, was_truncated, token_count).
    strategy='truncate': cut to max_tokens worth of words
    strategy='reject':   raise ValueError if over budget
    """
    tokens = estimate_tokens(text)
    if tokens <= max_tokens:
        return text, False, tokens
    if strategy == 'reject':
        raise ValueError(f'Input too long: {tokens} tokens (max {max_tokens})')
    max_words = int(max_tokens / 1.3)
    truncated = ' '.join(text.split()[:max_words])
    return truncated, True, max_tokens

def idempotent_load(df, table, key_col):
    df = df.drop_duplicates(subset=key_col, keep='last')
    tables = [r[0] for r in con.execute('SHOW TABLES').fetchall()]
    if table not in tables:
        con.execute(f'CREATE TABLE {table} AS SELECT * FROM df LIMIT 0')
    keys = ', '.join(str(k) for k in df[key_col].tolist())
    con.execute(f'DELETE FROM {table} WHERE {key_col} IN ({keys})')
    con.execute(f'INSERT INTO {table} SELECT * FROM df')
    return con.execute(f'SELECT COUNT(*) FROM {table}').fetchone()[0]

class DAG:
    def __init__(self, dag_id, schedule='@daily'):
        self.dag_id   = dag_id
        self.schedule = schedule
        self.tasks    = []
        self.on_failure_callback = None

    def task(self, fn):
        self.tasks.append(fn)
        return fn

    def run(self):
        results = {}
        for fn in self.tasks:
            try:
                results[fn.__name__] = fn(results)
            except Exception as exc:
                if self.on_failure_callback:
                    self.on_failure_callback(fn.__name__, str(exc))
                results[fn.__name__] = f'FAILED: {exc}'
        return results
