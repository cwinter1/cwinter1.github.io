# ── Simulation layer ──────────────────────────────────────────
# In real Airflow you import from airflow:
#   from airflow.decorators import dag, task
# Here we build equivalent Python classes.

class TaskResult:
    """Holds the outcome of a single task execution."""
    def __init__(self, task_id, status, return_value=None, error=None):
        self.task_id      = task_id
        self.status       = status       # 'success' | 'failed' | 'skipped'
        self.return_value = return_value
        self.error        = error
        self.started_at   = datetime.now().isoformat()

    def __repr__(self):
        return f'TaskResult({self.task_id}, {self.status})'


def task(func):
    """Decorator that wraps a function as an Airflow-style task."""
    @functools.wraps(func)
    def wrapper(*args, **kwargs):
        task_id = func.__name__
        try:
            result = func(*args, **kwargs)
            return TaskResult(task_id, 'success', return_value=result)
        except Exception as exc:
            return TaskResult(task_id, 'failed', error=str(exc))
    wrapper._is_task = True
    return wrapper

# ══════════════════════════════════════════════════════════════
#  CONCEPT 1 — WHAT AIRFLOW DOES
# ══════════════════════════════════════════════════════════════
# Apache Airflow is a workflow orchestration platform.
# It answers: "which tasks run, in what order, on what schedule?"
#
# Core concept: the DAG (Directed Acyclic Graph).
#   Directed:  tasks have a defined execution order (arrows point one way).
#   Acyclic:   no loops — a task cannot depend on itself or its own output.
#   Graph:     a set of tasks (nodes) connected by dependencies (edges).
#
# A DAG definition has:
#   dag_id       unique name used in the Airflow UI
#   schedule     cron expression or preset: '@daily', '@hourly', None (manual)
#   start_date   first date Airflow considers for scheduling
#   catchup      if True, Airflow runs past missed intervals on startup
#
# Airflow does NOT run your code — it schedules and monitors it.
# The actual computation happens in Workers (separate processes).
#
# EXAMPLE ──────────────────────────────────────────────────────

class DAG:
    """
    Simulation of an Airflow DAG.
    In real Airflow: @dag(dag_id='...', schedule='@daily', start_date=...)
    """
    def __init__(self, dag_id, schedule, start_date, catchup=False, description=''):
        self.dag_id      = dag_id
        self.schedule    = schedule
        self.start_date  = start_date
        self.catchup     = catchup
        self.description = description
        self._tasks      = []   # list of (task_id, callable)

    def register(self, task_fn):
        """Register a task function with this DAG."""
        self._tasks.append(task_fn)
        return task_fn

    def run(self, context=None):
        """
        Execute all tasks in registration order.
        Returns a run summary dict.
        """
        context = context or {}
        started = datetime.now()
        results = []
        xcom    = {}   # shared data store between tasks

        for fn in self._tasks:
            r = fn(xcom)
            xcom[r.task_id] = r.return_value
            results.append(r)
            if r.status == 'failed':
                break   # stop on first failure (like Airflow default)

        failed = [r for r in results if r.status == 'failed']
        return {
            'dag_id':           self.dag_id,
            'schedule':         self.schedule,
            'started_at':       started.isoformat(),
            'completed_at':     datetime.now().isoformat(),
            'tasks_run':        len(results),
            'tasks_succeeded':  len([r for r in results if r.status == 'success']),
            'tasks_failed':     len(failed),
            'status':           'FAILED' if failed else 'SUCCESS',
            'errors':           [r.error for r in failed],
        }

#  CONCEPT 2 — TASKFLOW API: @dag AND @task DECORATORS
# ══════════════════════════════════════════════════════════════
# The TaskFlow API (Airflow 2.0+) uses Python decorators.
# Instead of using Operators (PythonOperator, BashOperator),
# you write plain Python functions and decorate them with @task.
#
# In real Airflow:
#   from airflow.decorators import dag, task
#
#   @dag(dag_id='my_pipeline', schedule='@daily', start_date=datetime(2024,1,1))
#   def my_pipeline():
#       @task
#       def extract(): return [1, 2, 3]
#
#       @task
#       def transform(data): return [x * 2 for x in data]
#
#       transformed = transform(extract())
#   my_pipeline()
#
# Key rules for @task functions:
#   - Return a value to pass it downstream (XCom push happens automatically)
#   - Accept one argument to receive upstream output (XCom pull)
#   - Keep tasks small — each does ONE thing
#   - No side effects between tasks (tasks are isolated)
#
# EXAMPLE ──────────────────────────────────────────────────────

import pandas as pd

# Build a 3-task ETL DAG using our simulation layer

    dag_id     = 'etl_example',
    schedule   = '@daily',
    start_date = datetime(2024, 1, 1),
)

@etl_dag.register
@task
def extract(xcom):
    """Pull raw rows from source (simulated)."""
    raw = [
        {'order_id': 1, 'product': 'Widget A', 'amount':  120.50},
        {'order_id': 2, 'product': 'Widget B', 'amount':   85.00},
        {'order_id': 3, 'product': 'Widget A', 'amount':  210.00},
    ]
    print(f'  extract: pulled {len(raw)} rows')
    return raw

@etl_dag.register
@task
def transform(xcom):
    """Apply business rules to the raw rows."""
    raw = xcom.get('extract', [])
    df  = pd.DataFrame(raw)
    df['amount_gbp'] = (df['amount'] * 0.79).round(2)
    df['loaded_at']  = datetime.now().date().isoformat()
    print(f'  transform: converted {len(df)} rows to GBP')
    return df.to_dict('records')

@etl_dag.register
@task
def load(xcom):
    """Write transformed rows to destination (simulated)."""
    rows = xcom.get('transform', [])
    print(f'  load: wrote {len(rows)} rows to warehouse')
    return {'rows_loaded': len(rows)}

# ══════════════════════════════════════════════════════════════
#  CONCEPT 3 — DAG RUN LIFECYCLE
# ══════════════════════════════════════════════════════════════
# Each time Airflow triggers a DAG it creates a DAG Run.
# A DAG Run moves through these states:
#
#   queued     -> Airflow has scheduled the run, waiting for a free worker slot
#   running    -> at least one task is executing right now
#   success    -> all tasks completed with status 'success'
#   failed     -> at least one task finished with status 'failed'
#   up_for_retry -> task failed, retries remain, will be re-queued
#
# Each Task Instance inside a DAG Run has its own state:
#   scheduled  -> waiting for dependencies to clear
#   running    -> executing right now
#   success    -> finished without exception
#   failed     -> raised an exception
#   skipped    -> upstream returned False (ShortCircuit, BranchOperator)
#   upstream_failed -> a dependency failed so this task never ran
#
# The run_id in Airflow encodes the trigger time:
#   scheduled__2024-01-15T06:00:00+00:00
#   manual__2024-01-15T08:32:11+00:00
#
# EXAMPLE ──────────────────────────────────────────────────────

def simulate_lifecycle(dag, inject_failure_at=None):
    """
    Simulate a DAG run printing state transitions at each step.
    inject_failure_at: task index (0-based) to force a failure for demo.
    """
    print(f'[queued]   DAG run for "{dag.dag_id}" is queued')
    print(f'[running]  Starting DAG run')

    xcom    = {}
    results = []
    for i, fn in enumerate(dag._tasks):
        task_id = fn.__wrapped__.__name__ if hasattr(fn, '__wrapped__') else fn.__name__
        if inject_failure_at is not None and i == inject_failure_at:
            r = TaskResult(task_id, 'failed', error='Simulated failure')
            print(f'  [failed]   task: {task_id}  error: {r.error}')
            results.append(r)
            # remaining tasks become upstream_failed
            for remaining in dag._tasks[i+1:]:
                rid = remaining.__wrapped__.__name__ if hasattr(remaining, '__wrapped__') else remaining.__name__
                print(f'  [upstream_failed] task: {rid}')
            break
        else:
            r = fn(xcom)
            xcom[r.task_id] = r.return_value
            results.append(r)
            print(f'  [success]  task: {r.task_id}')

    dag_status = 'success' if all(r.status == 'success' for r in results) else 'failed'
    print(f'[{dag_status}]   DAG run complete')
    return dag_status

# ══════════════════════════════════════════════════════════════
#  CONCEPT 1 — XCOM: PASSING DATA BETWEEN TASKS
# ══════════════════════════════════════════════════════════════
# XCom (Cross-Communication) is Airflow's mechanism for tasks to
# share small amounts of data with each other.
#
# How it works:
#   task_a returns a value            --> Airflow stores it as xcom_push
#   task_b receives task_a's result   --> Airflow retrieves it as xcom_pull
#
# In real Airflow (TaskFlow API):
#   @task
#   def extract(): return {'rows': 42}         # auto push
#
#   @task
#   def load(data): print(data['rows'])         # auto pull
#
#   load(extract())   # wire them together
#
# Without TaskFlow (old API):
#   ti.xcom_push(key='row_count', value=42)
#   ti.xcom_pull(task_ids='extract', key='row_count')
#
# XCom size limit:
#   Data is stored in the Airflow metadata database (Postgres/MySQL).
#   Keep XCom values small (< 48 KB). For large DataFrames use:
#     - MinIO / S3 path returned as a string
#     - Custom XCom backend (Airflow 2.x supports this)
#
# EXAMPLE ──────────────────────────────────────────────────────

class XComStore:
    """
    Simulation of Airflow's XCom storage.
    In real Airflow this is backed by the metadata database.
    Key pattern: dag_run_id + task_id + key
    """
    def __init__(self):
        self._store = {}   # {(task_id, key): value}

    def push(self, task_id, value, key='return_value'):
        self._store[(task_id, key)] = value
        print(f'  xcom_push  task={task_id}  key={key}  value={repr(value)[:60]}')

    def pull(self, task_id, key='return_value'):
        val = self._store.get((task_id, key))
        print(f'  xcom_pull  task={task_id}  key={key}  -> {repr(val)[:60]}')
        return val


# Wire three tasks: extract -> enrich -> summarise
xcom = XComStore()

def extract_orders():
    orders = [{'id': 1, 'amount': 100}, {'id': 2, 'amount': 250}, {'id': 3, 'amount': 75}]
    xcom.push('extract_orders', orders)

def enrich_orders():
    orders = xcom.pull('extract_orders')
    for o in orders:
        o['vat'] = round(o['amount'] * 0.2, 2)
    xcom.push('enrich_orders', orders)

def summarise_orders():
    orders = xcom.pull('enrich_orders')
    total  = sum(o['amount'] + o['vat'] for o in orders)
    xcom.push('summarise_orders', {'total_with_vat': round(total, 2)})
    return total

# ══════════════════════════════════════════════════════════════
#  CONCEPT 2 — SENSORS: WAITING FOR A CONDITION
# ══════════════════════════════════════════════════════════════
# A Sensor is a special Airflow task that polls until a condition is True.
# It blocks downstream tasks until the condition is met.
#
# Common built-in sensors:
#   FileSensor        -- wait until a file appears at a path
#   S3KeySensor       -- wait until a key appears in S3 / MinIO
#   HttpSensor        -- wait until an HTTP endpoint returns 200
#   ExternalTaskSensor -- wait until a task in ANOTHER DAG succeeds
#   SqlSensor         -- wait until a SQL query returns rows
#
# Sensor parameters:
#   poke_interval  seconds between each check (default 60)
#   timeout        seconds before sensor times out and raises AirflowSensorTimeout
#   mode           'poke' (holds worker slot) or 'reschedule' (releases slot)
#
# Use mode='reschedule' for sensors that wait longer than a few minutes.
# This frees the worker slot between checks.
#
# EXAMPLE ──────────────────────────────────────────────────────

class FileSensor:
    """
    Simulation of Airflow's FileSensor.
    In real Airflow:
        FileSensor(task_id='wait_for_file', filepath='/data/orders.csv',
                   poke_interval=30, timeout=600, mode='reschedule')
    """
    def __init__(self, task_id, filepath, poke_interval=1, timeout=5):
        self.task_id       = task_id
        self.filepath      = filepath
        self.poke_interval = poke_interval
        self.timeout       = timeout

    def poke(self, available_files):
        """Return True if filepath is in the available_files list."""
        return self.filepath in available_files

    def run(self, file_stream):
        """
        Poll file_stream (a list of lists, one per tick) until
        filepath appears or timeout is reached.
        Returns 'success' or raises TimeoutError.
        """
        for tick, available_files in enumerate(file_stream):
            elapsed = tick * self.poke_interval
            if elapsed > self.timeout:
                raise TimeoutError(f'FileSensor timed out after {self.timeout}s')
            print(f'  [poke {tick+1}] checking for "{self.filepath}" ... ', end='')
            if self.poke(available_files):
                print('FOUND')
                return 'success'
            print('not found')
        raise TimeoutError(f'FileSensor timed out — file never appeared')

#  CONCEPT 3 — TASK DEPENDENCIES
# ══════════════════════════════════════════════════════════════
# In Airflow, dependencies define execution order.
#
# Three ways to set dependencies:
#
#   1. Bitshift operator (most common):
#        extract >> transform >> load
#        sensor >> [task_a, task_b]   # fan-out (parallel)
#        [task_a, task_b] >> merge    # fan-in
#
#   2. Explicit methods:
#        transform.set_upstream(extract)     # same as extract >> transform
#        transform.set_downstream(load)      # same as transform >> load
#
#   3. TaskFlow API (Day 1):
#        result = transform(extract())       # dependency implied by data flow
#
# Trigger rules (what makes a task run):
#   all_success   (default) -- all parents succeeded
#   all_done      -- all parents finished, regardless of status
#   one_success   -- at least one parent succeeded
#   none_failed   -- no parents failed (skipped is OK)
#
# EXAMPLE ──────────────────────────────────────────────────────

class TaskNode:
    """
    Simple simulation of an Airflow task node with >> dependency wiring.
    """
    def __init__(self, task_id, fn):
        self.task_id    = task_id
        self.fn         = fn
        self.upstreams  = []
        self.downstreams= []

    def __rshift__(self, other):
        """task_a >> task_b — task_b depends on task_a."""
        if isinstance(other, list):
            for o in other:
                o.upstreams.append(self)
                self.downstreams.append(o)
        else:
            other.upstreams.append(self)
            self.downstreams.append(other)
        return other

    def __repr__(self):
        ups  = [u.task_id for u in self.upstreams]
        downs= [d.task_id for d in self.downstreams]
        return f'TaskNode({self.task_id}, upstream={ups}, downstream={downs})'


def run_graph(entry_nodes, xcom):
    """Topological execution: run all reachable tasks starting from entry_nodes."""
    from collections import deque
    visited = set()
    queue   = deque(entry_nodes)
    results = {}

    while queue:
        node = queue.popleft()
        if node.task_id in visited:
            continue
        # Check all upstreams have succeeded
        if not all(u.task_id in results for u in node.upstreams):
            queue.append(node)   # re-queue, not ready yet
            continue
        visited.add(node.task_id)
        r = node.fn(xcom)
        xcom[node.task_id] = r.return_value
        results[node.task_id] = r
        print(f'  [{r.status}]  {node.task_id}')
        for d in node.downstreams:
            queue.append(d)

    return results


# Build a fan-out / fan-in DAG:
#   extract >> [validate, enrich] >> merge
@task
def t_extract(x):  return ['row1', 'row2', 'row3']

n_extract  = TaskNode('t_extract',  t_extract)

# ══════════════════════════════════════════════════════════════

#   postgres, mysql, s3, http, google_cloud_platform, slack, ftp ...

#  CONCEPT 2 — VARIABLES: RUNTIME CONFIG IN AIRFLOW
# ══════════════════════════════════════════════════════════════
# Airflow Variables are key-value pairs stored in the metadata database.
# Use them for configuration that may change between runs but does not
# contain credentials (use Connections for secrets).
#
# In real Airflow:
#   from airflow.models import Variable
#   batch_size = Variable.get('batch_size', default_var='1000')
#   config     = Variable.get('pipeline_config', deserialize_json=True)
#
# Good uses for Variables:
#   - batch_size, max_retries, alert_email
#   - feature flags ('enable_backfill': true)
#   - target environment ('env': 'staging')
#
# Anti-patterns:
#   - Storing passwords (use Connections or a secrets backend)
#   - Storing large datasets (Variable values are small strings)
#   - Reading Variables at module level (causes issues at parse time)
#
# EXAMPLE ──────────────────────────────────────────────────────

class VariableStore:
    """
    Simulation of Airflow's Variable store.
    In real Airflow: Variable.get('key', default_var='fallback')
    """
    def __init__(self):
        self._vars = {}

    def set(self, key, value):
        # JSON-encode dict/list values, store strings as-is
        self._vars[key] = json.dumps(value) if not isinstance(value, str) else value

    def get(self, key, default_var=None, deserialize_json=False):
        raw = self._vars.get(key, default_var)
        if deserialize_json and raw is not None:
            try:
                return json.loads(raw)
            except (json.JSONDecodeError, TypeError):
                return raw
        return raw

#  CONCEPT 3 — HOOKS: TYPED CONNECTION HELPERS
# ══════════════════════════════════════════════════════════════
# A Hook wraps a Connection and provides typed methods for a specific system.
# Instead of writing raw connection code, tasks call hook methods.
#
# In real Airflow:
#   from airflow.providers.postgres.hooks.postgres import PostgresHook
#   hook = PostgresHook(postgres_conn_id='postgres_prod')
#   rows = hook.get_records('SELECT dept, COUNT(*) FROM employees GROUP BY dept')
#
# Benefits:
#   - Credentials never appear in task code
#   - Common operations (get_records, get_pandas_df, run) reusable
#   - Connection can be swapped (dev/prod) by changing the conn_id variable
#
# Commonly used hooks:
#   PostgresHook    get_records(), get_pandas_df(), run()
#   S3Hook          load_file(), download_file(), list_keys()
#   HttpHook        run(), get_conn()
#   SlackHook       call_load_endpoint()
#
# EXAMPLE ──────────────────────────────────────────────────────

import pandas as pd

class PostgresHook:
    """
    Simulation of Airflow's PostgresHook.
    Backed by DuckDB (already populated with employees table).

    In real Airflow:
        from airflow.providers.postgres.hooks.postgres import PostgresHook
        hook = PostgresHook(postgres_conn_id='postgres_prod')
    """
    def __init__(self, postgres_conn_id, conn_store=None):
        self.postgres_conn_id = postgres_conn_id
        # In real Airflow, the hook reads credentials from the connection store
        if conn_store:
            c = conn_store.get_connection(postgres_conn_id)
            print(f'  [hook] connected via {c.get_uri()}')

    def get_records(self, sql, parameters=None):
        """Return a list of tuples (same interface as real PostgresHook)."""
        if parameters:
            result = con.execute(sql, parameters).fetchall()
        else:
            result = con.execute(sql).fetchall()
        return result

    def get_pandas_df(self, sql, parameters=None):
        """Return a DataFrame (same interface as real PostgresHook)."""
        if parameters:
            return con.execute(sql, parameters).df()
        return con.execute(sql).df()

    def run(self, sql, parameters=None):
        """Execute a statement with no return (INSERT, UPDATE, etc.)."""
        if parameters:
            con.execute(sql, parameters)
        else:
            con.execute(sql)

# ══════════════════════════════════════════════════════════════
#  CONCEPT 4 — SECURITY GATE: DAG SECURITY GATE (CIA: C+A)
# ══════════════════════════════════════════════════════════════
# Analogy: the data center access badge system — every engineer has a
# named badge with a role, expired badges are refused, and no one can
# prop the door open with a rock (shell=True).
#
# CIA focus: Confidentiality + Availability — DAG ownership must be
# declared; connections must not store plaintext default passwords;
# shell injection in BashOperator tasks must be blocked.
#
# Checks:
#   1. dag_config must have an 'owner' key — no orphaned DAGs
#   2. Each connection: password must not be empty or equal the conn_id
#   3. Tasks with bash_command containing '{{ ds }}' and shell=True are flagged
#
# EXAMPLE ──────────────────────────────────────────────────────────
from dataclasses import dataclass, field


class MissingOwnerError(Exception):
    pass


class WeakConnectionError(Exception):
    pass


class ShellInjectionRiskError(Exception):
    pass


@dataclass
class DagSecurityReport:
    passed: bool
    violations: list = field(default_factory=list)


def dag_security_gate(dag_config: dict, connections: list) -> DagSecurityReport:
    if "owner" not in dag_config:
        raise MissingOwnerError(
            f"DAG '{dag_config.get('dag_id', '?')}' has no owner declared"
        )
    for conn in connections:
        conn_id = conn.get("conn_id", "")
        password = conn.get("password", "")
        if password == "" or password == conn_id:
            raise WeakConnectionError(
                f"Connection '{conn_id}' has empty or default password"
            )
    for task in dag_config.get("tasks", []):
        bash_cmd = task.get("bash_command", "")
        if "{{ ds }}" in bash_cmd and task.get("shell") is True:
            raise ShellInjectionRiskError(
                f"Task '{task.get('task_id')}' has shell=True + template variable"
            )
    return DagSecurityReport(passed=True)

#  CONCEPT 1 — DYNAMIC TASK MAPPING: expand()
# ══════════════════════════════════════════════════════════════
# Dynamic task mapping (Airflow 2.3+) lets you generate tasks at runtime
# based on a list that may not be known until the DAG runs.
#
# In real Airflow (TaskFlow API):
#   @task
#   def process_file(filename: str) -> dict:
#       ...do something with the file...
#       return {'file': filename, 'rows': 1234}
#
#   # Fan out: one task instance per filename
#   results = process_file.expand(filename=file_list)
#
# Each call is an independent Task Instance in the Airflow UI.
# They can run in parallel (subject to worker slot availability).
#
# Airflow also supports partial() to fix some args and expand others:
#   process_file.partial(mode='strict').expand(filename=file_list)
#
# Use expand() when:
#   - The list size is determined by upstream data
#   - Each element can be processed independently
#   - You want retry at the individual-element level
#
# EXAMPLE ──────────────────────────────────────────────────────

def process_file(filename):
    """
    Simulate processing one file.
    In real Airflow this would be decorated with @task.
    """
    rows = random.randint(100, 5000)
    size_kb = rows * 0.5
    print(f'  processed  {filename:<35}  {rows:>4} rows  ({size_kb:.0f} KB)')
    return {'file': filename, 'rows': rows, 'size_kb': size_kb}


def expand(fn, items):
    """
    Simulation of Airflow's .expand() -- runs fn once per item.
    In real Airflow: fn.expand(arg=items)
    Results are collected in order (same as Airflow mapped task results).
    """
    return [fn(item) for item in items]

#  CONCEPT 2 — PARALLEL VS SEQUENTIAL EXECUTION
# ══════════════════════════════════════════════════════════════
# Airflow can run tasks in parallel when they have no dependency on each other.
# Concurrency is controlled at three levels:
#
#   parallelism         global: max concurrent task instances across ALL DAGs
#   dag max_active_tasks: max concurrent tasks within ONE DAG run
#   pool slots          logical grouping (e.g. 'db_pool': 3 slots max)
#
# When to use parallel:
#   - Independent file processing (different regions, different dates)
#   - Calls to different APIs with no shared state
#   - Backfill across many historical dates
#
# When to use sequential:
#   - Task B needs Task A's output
#   - Database writes that must maintain order
#   - Rate-limited APIs where parallel calls would be throttled
#
# Sequential pattern (chain): extract >> transform >> load
# Parallel pattern (fan-out): extract >> [validate, enrich, profile] >> merge
#
# EXAMPLE ──────────────────────────────────────────────────────

from concurrent.futures import ThreadPoolExecutor

def timed_task(name, duration_ms):
    """Simulate a task that takes duration_ms milliseconds."""
    start = datetime.now()
    time.sleep(duration_ms / 1000)
    elapsed = (datetime.now() - start).total_seconds() * 1000
    return {'task': name, 'duration_ms': round(elapsed, 0)}

tasks = [
    ('validate_emea', 80),
    ('validate_amer', 90),
    ('validate_apac', 70),
]

# Sequential

seq_results = [timed_task(n, d) for n, d in tasks]

# Parallel (threads simulate Airflow workers)

print(f'Parallel:    {par_total:.0f} ms total')
print(f'Speedup:     {seq_total/par_total:.1f}x')

#  CONCEPT 3 — ERROR HANDLING: RETRIES AND CALLBACKS
# ══════════════════════════════════════════════════════════════
# Airflow tasks can be configured to retry automatically on failure.
#
# In real Airflow (Python Operator or @task decorator):
#   @task(retries=3, retry_delay=timedelta(minutes=5))
#   def flaky_api_call(): ...
#
# Or with exponential backoff:
#   @task(retries=3, retry_delay=timedelta(seconds=30),
#         retry_exponential_backoff=True, max_retry_delay=timedelta(minutes=10))
#   def flaky_api_call(): ...
#
# on_failure_callback:
#   A Python function called when a task fails all retries.
#   In real Airflow: receives a context dict with dag, task, exception info.
#
#   @task(on_failure_callback=alert_slack)
#   def risky_task(): ...
#
#   def alert_slack(context):
#       error = context['exception']
#       send_slack_message(f'Task failed: {error}')
#
# EXAMPLE ──────────────────────────────────────────────────────

def with_retry(fn, max_retries=3, base_delay=0.05):
    """
    Run fn with exponential backoff retry.
    Simulation of Airflow's retry_exponential_backoff=True.
    base_delay in seconds (kept tiny for demo speed).
    """
    last_error = None
    for attempt in range(1, max_retries + 1):
        try:
            result = fn()
            print(f'  attempt {attempt}: SUCCESS')
            return result
        except Exception as exc:
            last_error = exc
            delay = base_delay * (2 ** (attempt - 1))
            print(f'  attempt {attempt}: FAILED ({exc}) -- retry in {delay:.2f}s')
            time.sleep(delay)
    raise RuntimeError(f'Task failed after {max_retries} retries: {last_error}')


def on_failure_callback(context):
    print(f'  [on_failure_callback] task={context["task"]}  error={context["error"]}')


# Task that fails twice then succeeds
attempt_counter = {'n': 0}

def flaky_api_call():
    attempt_counter['n'] += 1
    if attempt_counter['n'] < 3:
        raise ConnectionError('API timeout')
    return {'status': 200, 'records': 42}


print('--- Task with retry ---')
try:

    print(f'  result: {result}')
except RuntimeError as exc:
    on_failure_callback({'task': 'flaky_api_call', 'error': str(exc)})

def bq_load(df, table, write_disposition='WRITE_TRUNCATE'):
    tables = [r[0] for r in con.execute('SHOW TABLES').fetchall()]
    if write_disposition == 'WRITE_TRUNCATE' or table not in tables:
        con.execute(f'CREATE OR REPLACE TABLE {table} AS SELECT * FROM df')
    else:
        con.execute(f'INSERT INTO {table} SELECT * FROM df')
    return con.execute(f'SELECT COUNT(*) FROM {table}').fetchone()[0]

def dbt_run(name, sql, materialized='view'):
    if materialized == 'table':
        con.execute(f'CREATE OR REPLACE TABLE {name} AS {sql}')
    else:
        con.execute(f'CREATE OR REPLACE VIEW {name} AS {sql}')
    return con.execute(f'SELECT COUNT(*) FROM {name}').fetchone()[0]

def dbt_test_not_null(model, col):
    n = con.execute(f'SELECT COUNT(*) FROM {model} WHERE {col} IS NULL').fetchone()[0]
    return n == 0

def dbt_test_unique(model, col):
    n = con.execute(f"""SELECT COUNT(*) FROM (
        SELECT {col} FROM {model} GROUP BY {col} HAVING COUNT(*) > 1
    )""").fetchone()[0]
    return n == 0

def dbt_test_accepted_values(model, col, values):
    placeholders = ', '.join([f"'{v}'" for v in values])
    n = con.execute(f'SELECT COUNT(*) FROM {model} WHERE {col} NOT IN ({placeholders})').fetchone()[0]
    return n == 0

#  CONCEPT 1 — dbt + AIRFLOW: TWO TOOLS, ONE PIPELINE
# ══════════════════════════════════════════════════════════════
# dbt and Airflow solve different problems and work together:
#
#   Airflow handles:              dbt handles:
#   ─────────────────────         ──────────────────────────
#   WHEN to run (schedule)        WHAT to compute (SQL models)
#   Data extraction (APIs, DBs)   Transformations (staging, marts)
#   File movement (S3, GCS)       Data quality tests (schema.yml)
#   Notifications (Slack, email)  Documentation (dbt docs)
#   Retry / failure logic         Dependency graph within dbt
#
# Typical production pipeline:
#   1. Airflow extracts from source API -> loads raw table
#   2. Airflow calls: dbt run --select staging+
#   3. Airflow calls: dbt test --select staging+
#   4. Airflow calls: dbt run --select mart+
#   5. On failure: Airflow on_failure_callback sends alert
#
# Two integration patterns:
#   A. BashOperator / subprocess:
#        BashOperator(task_id='dbt_run', bash_command='dbt run --profiles-dir .')
#
#   B. astronomer-cosmos (recommended for large projects):
#        DbtTaskGroup() maps each dbt model to its own Airflow task
#        Full lineage visible in Airflow UI
#
# EXAMPLE ──────────────────────────────────────────────────────

# Simulate a DbtRunResult as returned by subprocess / cosmos
from dataclasses import dataclass

@dataclass
class DbtRunResult:
    model         : str
    status        : str   # 'success' | 'error' | 'skip'
    rows_affected : int
    duration_ms   : int

    def to_dict(self):
        return {
            'model': self.model, 'status': self.status,
            'rows_affected': self.rows_affected, 'duration_ms': self.duration_ms
        }


class DbtRunOperator:
    """
    Simulation of the Airflow DbtRunOperator (from astronomer-cosmos or similar).

    In real Airflow (BashOperator approach):
        DbtRunOperator(
            task_id='dbt_run_staging',
            models=['stg_sales'],
            dbt_bin='/usr/local/bin/dbt',
            profiles_dir='/opt/airflow/dbt',
        )

    Real subprocess call (BashOperator alternative):
        import subprocess
        result = subprocess.run(
            ['dbt', 'run', '--select', model_name, '--profiles-dir', '.'],
            capture_output=True, text=True
        )
        if result.returncode != 0:
            raise RuntimeError(result.stderr)
    """
    def __init__(self, task_id, model, sql, materialized='view'):
        self.task_id      = task_id
        self.model        = model
        self.sql          = sql
        self.materialized = materialized

    def execute(self):
        start = datetime.now()
        rows  = dbt_run(self.model, self.sql, self.materialized)
        ms    = int((datetime.now() - start).total_seconds() * 1000)
        return DbtRunResult(self.model, 'success', rows, max(ms, 1))

class DbtTestRunner:
    """Simulation of dbt test command run from Airflow."""

    def __init__(self):
        self.tests   = []    # list of (name, fn)
        self.results = []

    def add_not_null(self, model, col):
        self.tests.append((f'{model}.{col}.not_null',
                           lambda m=model, c=col: dbt_test_not_null(m, c)))

    def add_unique(self, model, col):
        self.tests.append((f'{model}.{col}.unique',
                           lambda m=model, c=col: dbt_test_unique(m, c)))

    def add_accepted_values(self, model, col, values):
        self.tests.append((f'{model}.{col}.accepted_values',
                           lambda m=model, c=col, v=values: dbt_test_accepted_values(m, c, v)))

    def run(self):
        for name, fn in self.tests:
            passed = fn()
            status = 'PASS' if passed else 'FAIL'
            self.results.append({'test': name, 'status': status})
            print(f'  [{status}]  {name}')
        failed = [r for r in self.results if r['status'] == 'FAIL']
        return len(failed) == 0

def on_failure_callback(context):
    """
    Called when any task in the DAG fails all retries.

        f'Airflow pipeline failed.\n'
        f'DAG: {context["dag_id"]}\n'
        f'Task: {context["task_id"]}\n'
        f'Error: {context["error"]}\n'
        f'Explain what went wrong and suggest a fix in 3 sentences.'

    explanation = call_llm(prompt)

con = duckdb.connect(':memory:')

def ref(m): return m
def source(s, t): return f'{s}__{t}'

def bq_load(df, table, write_disposition='WRITE_TRUNCATE'):
    tables = [r[0] for r in con.execute('SHOW TABLES').fetchall()]
    if write_disposition == 'WRITE_TRUNCATE' or table not in tables:
        con.execute(f'CREATE OR REPLACE TABLE {table} AS SELECT * FROM df')
    else:
        con.execute(f'INSERT INTO {table} SELECT * FROM df')
    return con.execute(f'SELECT COUNT(*) FROM {table}').fetchone()[0]

def dbt_run(name, sql, materialized='view'):
    if materialized == 'table':
        con.execute(f'CREATE OR REPLACE TABLE {name} AS {sql}')
    else:
        con.execute(f'CREATE OR REPLACE VIEW {name} AS {sql}')
    return con.execute(f'SELECT COUNT(*) FROM {name}').fetchone()[0]

def dbt_test_not_null(model, col):
    return con.execute(f'SELECT COUNT(*) FROM {model} WHERE {col} IS NULL').fetchone()[0] == 0

def dbt_test_unique(model, col):
    return con.execute(f"""
        SELECT COUNT(*) FROM (
            SELECT {col} FROM {model} GROUP BY {col} HAVING COUNT(*) > 1
        )""").fetchone()[0] == 0

def dbt_test_positive(model, col):
    return con.execute(f'SELECT COUNT(*) FROM {model} WHERE {col} <= 0').fetchone()[0] == 0
