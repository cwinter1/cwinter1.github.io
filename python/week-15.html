# ══════════════════════════════════════════════════════════════
#  CONCEPT 1 — BIGQUERY ARCHITECTURE
# ══════════════════════════════════════════════════════════════
# BigQuery is a serverless, columnar data warehouse on GCP.
# Key design choices:
#
#   Serverless:      No servers to manage. You pay per query (TB scanned) or by slot.
#   Columnar:        Data stored by column, not by row. Reads only the columns your query uses.
#                    A query on 1 column of a 100-column table reads ~1% of the data.
#   Separated:       Storage (Colossus) is decoupled from compute (Dremel).
#                    Scale storage and compute independently.
#
# Pricing model (on-demand):
#   $5 per TB of data scanned
#   First 1 TB per month is free
#
# Key objects:
#   Project   -> top-level billing unit (e.g. my-company-prod)
#   Dataset   -> namespace for tables (e.g. analytics, raw, staging)
#   Table     -> rows and columns (like a SQL table)
#   View      -> saved query (no storage cost, query runs on the fly)
#
# EXAMPLE ──────────────────────────────────────────────────────────

def estimate_bq_cost(bytes_scanned, price_per_tb=5.0):
    """Estimate BigQuery on-demand query cost."""
    tb_scanned = bytes_scanned / (1024 ** 4)
    cost = tb_scanned * price_per_tb
    free_tb = 1.0
    billable_tb = max(0, tb_scanned - free_tb)
    billable_cost = billable_tb * price_per_tb
    return {
        'bytes_scanned': bytes_scanned,
        'tb_scanned':    round(tb_scanned, 6),
        'estimated_cost_usd': round(billable_cost, 4),
        'note': 'First 1TB/month free on on-demand pricing'
    }

# Dry-run examples: how much would these queries cost?
scenarios = [
    ('SELECT * FROM events WHERE user_id = 123',     500_000_000),       # 500 MB
    ('SELECT COUNT(*) FROM events',                 50_000_000_000),     # 50 GB
    ('Full table scan on 2TB events table',      2_000_000_000_000),     # 2 TB
]

for query, bytes_scanned in scenarios:
    est = estimate_bq_cost(bytes_scanned)
    print(f'  Query: {query[:50]}')
    print(f'    Scans: {est["tb_scanned"]} TB  |  Cost: ${est["estimated_cost_usd"]:.4f}')
    print()

# ══════════════════════════════════════════════════════════════
#  CONCEPT 2 — PARTITIONING AND CLUSTERING
# ══════════════════════════════════════════════════════════════
# PARTITIONING: split table into segments by a date or integer column.
#   - A query with WHERE event_date = '2024-01' only scans that partition.
#   - 90% cost reduction is typical for time-filtered queries.
#   - Best for: time-series tables (events, logs, transactions)
#
# CLUSTERING: sort rows within each partition by one or more columns.
#   - A query with WHERE country = 'GB' skips blocks where country != 'GB'.
#   - Works best after partitioning. Reduces bytes scanned further.
#   - Best for: high-cardinality filter columns (country, product_id, user_id)
#
# In SQL (BigQuery DDL):
#   CREATE TABLE analytics.events
#   PARTITION BY DATE(event_timestamp)
#   CLUSTER BY country, event_type
#   AS SELECT ...
#
# EXAMPLE ──────────────────────────────────────────────────────────
def estimate_scan_with_optimization(total_tb, partitions, queried_partitions,
                                     cluster_selectivity=1.0):
    """Estimate bytes scanned with and without partitioning + clustering."""
    full_scan_tb       = total_tb
    partitioned_tb     = total_tb * (queried_partitions / partitions)
    clustered_tb       = partitioned_tb * cluster_selectivity
    return {
        'no_optimization':  round(full_scan_tb, 3),
        'partitioned':      round(partitioned_tb, 3),
        'partitioned_clustered': round(clustered_tb, 3),
    }

# events table: 365 daily partitions, querying 1 day, cluster selectivity 20%
result = estimate_scan_with_optimization(
    total_tb=10.0, partitions=365, queried_partitions=1, cluster_selectivity=0.20
)

for strategy, tb in result.items():
    cost = estimate_bq_cost(tb * 1024**4)['estimated_cost_usd']
    print(f'  {strategy:<30}  {tb:>6.3f} TB  ->  ${cost:.4f}')

# ─── SIMULATION MODE ──────────────────────────────────────────────────────────

# REAL MODE: replace BigQuerySimulator with:
#   from google.cloud import bigquery
#   client = bigquery.Client(project='my-project')

@dataclass
class QueryParameter:
    name:  str
    type_: str
    value: Any

class BigQuerySimulator:
    """DuckDB-backed simulation of the BigQuery Python client."""
    def __init__(self):
        self.con = duckdb.connect(':memory:')
        self._bytes_scanned = 0

    def query(self, sql, query_parameters=None):
        if query_parameters:
            for p in query_parameters:
                val = f"'{p.value}'" if p.type_ in ('STRING', 'DATE') else str(p.value)
                sql = sql.replace(f'@{p.name}', val)
        df = self.con.execute(sql).fetchdf()
        self._bytes_scanned += len(df) * 100  # simulate
        return df

    def load_table(self, table_name, df):
        self.con.execute(f'CREATE OR REPLACE TABLE {table_name} AS SELECT * FROM df')
        print(f'  Loaded {len(df)} rows into {table_name}')

import duckdb
import pandas as pd

class BigQuerySimulator:
    def __init__(self):
        self.con = duckdb.connect(':memory:')

    def load_table_from_dataframe(self, df, table_id, write_disposition='WRITE_TRUNCATE'):
        """Simulate BigQuery load job from a DataFrame."""
        tables = [r[0] for r in self.con.execute('SHOW TABLES').fetchall()]
        if write_disposition == 'WRITE_TRUNCATE' or table_id not in tables:
            self.con.execute(f'CREATE OR REPLACE TABLE {table_id} AS SELECT * FROM df')
        elif write_disposition == 'WRITE_APPEND':
            self.con.execute(f'INSERT INTO {table_id} SELECT * FROM df')
        elif write_disposition == 'WRITE_EMPTY':
            if table_id in tables:
                existing = self.con.execute(f'SELECT COUNT(*) FROM {table_id}').fetchone()[0]
                if existing > 0:
                    raise ValueError(f'WRITE_EMPTY failed: {table_id} already has {existing} rows')
            self.con.execute(f'CREATE OR REPLACE TABLE {table_id} AS SELECT * FROM df')
        n = self.con.execute(f'SELECT COUNT(*) FROM {table_id}').fetchone()[0]
        print(f'  [{write_disposition}]  {table_id}  -> {n} total rows')

    def query(self, sql):
        return self.con.execute(sql).fetchdf()

# ══════════════════════════════════════════════════════════════
#  CONCEPT 1 — WRITE DISPOSITIONS
# ══════════════════════════════════════════════════════════════
# WRITE_TRUNCATE -> DELETE all existing rows, then write new rows
#   Use for: full refresh tables (dimension tables, small lookups)
#   Risk:    if the load fails, table is empty
#
# WRITE_APPEND -> ADD new rows to existing rows (no delete)
#   Use for: event/log tables, incremental fact tables
#   Risk:    re-running creates duplicates unless you deduplicate
#
# WRITE_EMPTY -> fail if table already has rows
#   Use for: one-time migration, seed data loads
#
# EXAMPLE ──────────────────────────────────────────────────────────
products_v1 = pd.DataFrame([
    {'product_id': 1, 'name': 'Widget',  'price': 15.00},
    {'product_id': 2, 'name': 'Gadget',  'price': 39.99},
])

products_v2 = pd.DataFrame([

    {'product_id': 2, 'name': 'Gadget',  'price': 39.99},
    {'product_id': 3, 'name': 'Gizmo',   'price': 59.00},  # new product
])

print('After initial load:', client.query('SELECT * FROM dim_products').to_string())

# ══════════════════════════════════════════════════════════════
#  CONCEPT 2 — PRE/POST LOAD VALIDATION
# ══════════════════════════════════════════════════════════════
# Production rule: VALIDATE the DataFrame before writing to BigQuery.
# A bad load can corrupt a table that serves 50 dashboards.
#
# Pre-load checks (on the DataFrame):
#   - no nulls in required columns
#   - row count > 0
#   - column values in expected ranges
#
# Post-load checks (query BigQuery after loading):
#   - row count matches source DataFrame
#   - no nulls introduced during load
#
# EXAMPLE ──────────────────────────────────────────────────────────
def validated_load(client, df, table_id, required_cols, write_disposition='WRITE_TRUNCATE'):
    """Validate a DataFrame then load it to BigQuery."""
    # Pre-load checks
    assert len(df) > 0, 'DataFrame is empty — aborting load'
    for col in required_cols:
        null_count = df[col].isna().sum()
        assert null_count == 0, f'Pre-load FAIL: {col} has {null_count} null values'
    print(f'  Pre-load checks passed for {table_id} ({len(df)} rows)')

    # Load
    client.load_table_from_dataframe(df, table_id, write_disposition)

    # Post-load check
    bq_count = client.query(f'SELECT COUNT(*) AS n FROM {table_id}').iloc[0]['n']
    assert bq_count == len(df), f'Post-load FAIL: BQ has {bq_count} rows, expected {len(df)}'
    print(f'  Post-load check passed: {bq_count} rows confirmed in {table_id}')

clean_df = pd.DataFrame([
    {'order_id': 10, 'customer_id': 201, 'amount': 500.00},
    {'order_id': 11, 'customer_id': 202, 'amount': 120.00},
])

TABLE_REGISTRY = {
    'events': {
        'total_bytes': 500_000_000_000,
        'columns': {
            'event_id':        5_000_000_000,
            'user_id':         5_000_000_000,
            'event_type':      3_000_000_000,
            'event_timestamp': 5_000_000_000,
            'page_url':       50_000_000_000,
            'properties':    200_000_000_000,
            'raw_json':      220_000_000_000,
        }
    }
}

def dry_run(sql, table_registry=TABLE_REGISTRY):
    table = next((t for t in table_registry if t in sql.lower()), None)
    if not table:
        return {'bytes_scanned': 0, 'tb_scanned': 0.0, 'cost_usd': 0.0}
    meta = table_registry[table]
    if 'select *' in sql.lower():
        bytes_scanned = meta['total_bytes']
    else:
        bytes_scanned = sum(v for col, v in meta['columns'].items() if col in sql)
        if bytes_scanned == 0:
            bytes_scanned = meta['total_bytes']
    tb = bytes_scanned / (1024 ** 4)
    cost = max(0, tb - 1.0) * 5.0
    return {'bytes_scanned': bytes_scanned, 'tb_scanned': round(tb, 3), 'cost_usd': round(cost, 4)}

# ══════════════════════════════════════════════════════════════
#  CONCEPT 4 — SECURITY GATE: BIGQUERY QUERY SECURITY GATE (CIA: A)
# ══════════════════════════════════════════════════════════════
# Analogy: the power grid's circuit breaker — normal load flows freely,
# but a sudden spike beyond the rated capacity trips the breaker before
# it damages the infrastructure.
#
# CIA focus: Availability — runaway queries are a denial-of-service attack
# on the data platform's budget and slot capacity.
#
# Checks (in order):
#   1. SELECT * on a large table (> 1M rows) -> FullTableScanError
#   2. PII column referenced without a WHERE clause -> PIIFullScanError
#   3. Estimated cost exceeds max_cost_usd -> CostLimitError
#
# dry_run() is defined in the header cell above — reused here.
#
# EXAMPLE ──────────────────────────────────────────────────────────
class FullTableScanError(Exception):
    pass


class PIIFullScanError(Exception):
    pass


class CostLimitError(Exception):
    pass


def bq_security_gate(
    query: str,
    table_sizes: dict,
    pii_columns: list,
    cost_per_tb: float = 6.25,
    max_cost_usd: float = 10.0,
) -> dict:
    q_lower = query.lower()
    if "select *" in q_lower:
        for table, row_count in table_sizes.items():
            if table in q_lower and row_count > 1_000_000:
                raise FullTableScanError(
                    f"SELECT * on large table '{table}' ({row_count:,} rows)"
                )
    for col in pii_columns:
        if col in query and "WHERE" not in query.upper():
            raise PIIFullScanError(
                f"PII column '{col}' scanned without WHERE clause"
            )
    est = dry_run(query)
    if est["cost_usd"] > max_cost_usd:
        raise CostLimitError(
            f"Estimated ${est['cost_usd']:.4f} exceeds budget ${max_cost_usd:.2f}"
        )
    return {"status": "PASS", "estimated_cost_usd": est["cost_usd"]}

con = duckdb.connect(':memory:')

def ref(m): return m
def source(s, t): return f'{s}__{t}'

def bq_load(df, table, write_disposition='WRITE_TRUNCATE'):
    tables = [r[0] for r in con.execute('SHOW TABLES').fetchall()]
    if write_disposition == 'WRITE_TRUNCATE' or table not in tables:
        con.execute(f'CREATE OR REPLACE TABLE {table} AS SELECT * FROM df')
    elif write_disposition == 'WRITE_APPEND':
        con.execute(f'INSERT INTO {table} SELECT * FROM df')
    n = con.execute(f'SELECT COUNT(*) FROM {table}').fetchone()[0]
    return n

def dbt_run(name, sql, materialized='view'):
    if materialized == 'table':
        con.execute(f'CREATE OR REPLACE TABLE {name} AS {sql}')
    else:
        con.execute(f'CREATE OR REPLACE VIEW {name} AS {sql}')
    n = con.execute(f'SELECT COUNT(*) FROM {name}').fetchone()[0]
    return n

def dbt_test_not_null(model, col):
    n = con.execute(f'SELECT COUNT(*) FROM {model} WHERE {col} IS NULL').fetchone()[0]
    return n == 0

def dbt_test_unique(model, col):
    n = con.execute(f'SELECT COUNT(*) FROM (SELECT {col} FROM {model} GROUP BY {col} HAVING COUNT(*) > 1)').fetchone()[0]
    return n == 0

def dbt_test(model, col, test_type, values=None):
    if test_type == 'not_null':
        return con.execute(f'SELECT COUNT(*) FROM {model} WHERE {col} IS NULL').fetchone()[0] == 0
    elif test_type == 'unique':
        return con.execute(f'SELECT COUNT(*) FROM (SELECT {col} FROM {model} GROUP BY {col} HAVING COUNT(*) > 1)').fetchone()[0] == 0

        vals = ', '.join(f"'{v}'" for v in values)
        return con.execute(f'SELECT COUNT(*) FROM {model} WHERE {col} NOT IN ({vals})').fetchone()[0] == 0
    return False
