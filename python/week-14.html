# ══════════════════════════════════════════════════════════════
#  CONCEPT 1 — WHY INCREMENTAL?
# ══════════════════════════════════════════════════════════════
# For a table with 100M rows, a full refresh (table materialization) re-reads
# and re-writes all 100M rows on every dbt run. This is expensive.
#
# Incremental materialization:
#   - On first run:  CREATE TABLE AS SELECT (full)
#   - On later runs: only INSERT/UPDATE rows newer than the last run
#
# In dbt, the SQL model contains a Jinja block:
#   {% if is_incremental() %}
#       WHERE event_date > (SELECT MAX(event_date) FROM {{ this }})
#   {% endif %}
#
# {{ this }} refers to the existing target table.
# is_incremental() returns True on all runs EXCEPT the very first one.
#
# EXAMPLE ──────────────────────────────────────────────────────────

def run_incremental(con, target_table, source_table, watermark_col, select_sql):
    """Simulate a dbt incremental model run."""
    tables = [r[0] for r in con.execute("SHOW TABLES").fetchall()]
    if target_table not in tables:
        # First run: full load
        con.execute(f'CREATE TABLE {target_table} AS {select_sql}')
        n = con.execute(f'SELECT COUNT(*) FROM {target_table}').fetchone()[0]
        print(f'  [FULL LOAD]  {target_table}: {n} rows loaded')
    else:
        # Incremental run: only new rows
        max_val = con.execute(f'SELECT MAX({watermark_col}) FROM {target_table}').fetchone()[0]
        inc_sql = f'{select_sql} WHERE {watermark_col} > \'{max_val}\''
        con.execute(f'INSERT INTO {target_table} {inc_sql}')
        n = con.execute(f'SELECT COUNT(*) FROM {target_table}').fetchone()[0]
        print(f'  [INCREMENTAL] {target_table}: {n} total rows (watermark > {max_val})')

# Simulate event source
con.execute('''
    CREATE TABLE raw__events AS SELECT * FROM (VALUES
        (1, '2024-01-01', 'page_view', 'homepage'),
        (2, '2024-01-01', 'click',     'buy_button'),
        (3, '2024-01-02', 'page_view', 'pricing')
    ) AS t(event_id, event_date, event_type, page)

select_all = f"SELECT event_id, event_date, event_type, page FROM {source('raw','events')}"

# ══════════════════════════════════════════════════════════════
#  CONCEPT 2 — UNIQUE KEY AND MERGE
# ══════════════════════════════════════════════════════════════
# Pure INSERT fails when source data is updated (re-delivered with corrections).
# Solution: unique_key = dbt will MERGE instead of INSERT.
#
# dbt config:
#   {{ config(materialized='incremental', unique_key='order_id') }}
#
# With unique_key set:
#   - If order_id already exists in target -> UPDATE that row
#   - If order_id is new                  -> INSERT the row
#
# This makes the pipeline IDEMPOTENT: re-running it produces the same result.
#
# EXAMPLE ──────────────────────────────────────────────────────────
def upsert(con, target, source, unique_key, cols):
    """Simulate dbt incremental with unique_key (MERGE behavior)."""
    col_list = ', '.join(cols)
    update_set = ', '.join(f't.{c} = s.{c}' for c in cols if c != unique_key)
    tables = [r[0] for r in con.execute('SHOW TABLES').fetchall()]
    if target not in tables:
        con.execute(f'CREATE TABLE {target} AS SELECT {col_list} FROM {source}')
        print(f'  [FULL LOAD]   {target}')
    else:
        # DELETE existing matching rows then re-insert (DuckDB MERGE simulation)
        con.execute(f'DELETE FROM {target} WHERE {unique_key} IN (SELECT {unique_key} FROM {source})')
        con.execute(f'INSERT INTO {target} SELECT {col_list} FROM {source}')
        print(f'  [UPSERT]      {target}')

# ══════════════════════════════════════════════════════════════
#  CONCEPT 1 — SCD TYPE 2 AND dbt SNAPSHOTS
# ══════════════════════════════════════════════════════════════
# SCD = Slowly Changing Dimension. Type 2 = keep history of changes.
#
# Problem: a customer moves from 'silver' tier to 'gold' tier.
# If you just UPDATE the row, you lose the history.
# For historical analysis ("what tier was this customer in Q1?") you need history.
#
# SCD Type 2 adds two columns:
#   dbt_valid_from  -> date/time this version became active
#   dbt_valid_to    -> date/time this version was superseded (NULL = still current)
#
# dbt snapshot config (in snapshots/snap_customers.sql):
#   {% snapshot snap_customers %}
#   {{ config(target_schema='snapshots', unique_key='customer_id',
#             strategy='check', check_cols=['tier', 'email']) }}
#   SELECT customer_id, email, tier FROM {{ source('raw', 'customers') }}
#   {% endsnapshot %}
#
# EXAMPLE ──────────────────────────────────────────────────────────
def take_snapshot(con, snap_table, source_table, unique_key, check_cols, as_of_date):
    """Simulate dbt snapshot behavior."""
    tables = [r[0] for r in con.execute('SHOW TABLES').fetchall()]
    if snap_table not in tables:
        cols = ', '.join([f's.{c}' for c in [unique_key] + check_cols])
        con.execute(f'''
            CREATE TABLE {snap_table} AS
            SELECT {unique_key}, {', '.join(check_cols)},
                   CAST('{as_of_date}' AS DATE) AS dbt_valid_from,
                   CAST(NULL AS DATE)            AS dbt_valid_to
            FROM {source_table}
        ''')
        print(f'  [SNAPSHOT INIT]   {snap_table} — {as_of_date}')
    else:
        # Close out changed rows, insert new versions
        changed_keys = con.execute(f'''
            SELECT s.{unique_key}
            FROM {source_table} s
            JOIN {snap_table} t ON s.{unique_key} = t.{unique_key}
            WHERE t.dbt_valid_to IS NULL
              AND ({' OR '.join(f's.{c} != t.{c}' for c in check_cols)})
        ''').fetchall()
        for (key,) in changed_keys:
            con.execute(f"UPDATE {snap_table} SET dbt_valid_to = '{as_of_date}' WHERE {unique_key} = {key} AND dbt_valid_to IS NULL")
            con.execute(f'''
                INSERT INTO {snap_table}
                SELECT {unique_key}, {', '.join(check_cols)},
                       CAST('{as_of_date}' AS DATE), CAST(NULL AS DATE)
                FROM {source_table} WHERE {unique_key} = {key}
            ''')
        print(f'  [SNAPSHOT UPDATE] {snap_table} — {as_of_date} ({len(changed_keys)} changes)')

con.execute('''
    CREATE TABLE raw__customers AS SELECT * FROM (VALUES
        (1, 'alice@x.com', 'silver'),
        (2, 'bob@x.com',   'bronze')
    ) AS t(customer_id, email, tier)
''')

take_snapshot(con, 'snap_customers', 'raw__customers', 'customer_id', ['email','tier'], '2024-01-01')

# Alice upgrades to gold
con.execute("UPDATE raw__customers SET tier = 'gold' WHERE customer_id = 1")

print(con.execute('SELECT * FROM snap_customers ORDER BY customer_id, dbt_valid_from').fetchdf())

# ══════════════════════════════════════════════════════════════
#  CONCEPT 4 — SECURITY GATE: SCD2 INTEGRITY CHECK (CIA: I)
# ══════════════════════════════════════════════════════════════
# Analogy: the notarized legal archive — every historical version of a
# record is sealed with a timestamp, and any gap or overlap in the
# timeline is grounds to invalidate the record.
#
# CIA focus: Integrity — historical records in SCD2 must be tamper-evident;
# overlapping date ranges are a data integrity failure.
#
# Checks per entity:
#   1. valid_from must never be null
#   2. For non-current rows (valid_to is not null), no two rows may overlap
#      i.e. row_a.valid_to must not exceed row_b.valid_from
#
# EXAMPLE ──────────────────────────────────────────────────────────
import pandas as pd
from dataclasses import dataclass


class OverlappingRangeError(Exception):
    pass


class NullDateError(Exception):
    pass


@dataclass
class IntegrityReport:
    passed: bool
    entity_count: int
    violation_count: int


def scd2_integrity_check(df, id_col, valid_from_col, valid_to_col) -> IntegrityReport:
    for entity_id, group in df.groupby(id_col):
        rows = group.sort_values(valid_from_col).reset_index(drop=True)
        for i, row in rows.iterrows():
            if pd.isna(row[valid_from_col]):
                raise NullDateError(
                    f"entity {entity_id} row {i}: {valid_from_col} is null"
                )
            if pd.notna(row[valid_to_col]):
                for j, other in rows.iterrows():
                    if j <= i:
                        continue
                    if (pd.notna(other[valid_from_col])
                            and row[valid_to_col] > other[valid_from_col]):
                        raise OverlappingRangeError(
                            f"entity {entity_id}: rows {i} and {j} overlap"
                        )
    return IntegrityReport(
        passed=True,
        entity_count=df[id_col].nunique(),
        violation_count=0,
    )


# Clean SCD2 — passes
clean = pd.DataFrame([
    {"emp_id": 401, "grade": "A", "valid_from": "2024-01-01", "valid_to": "2024-06-01"},
    {"emp_id": 401, "grade": "B", "valid_from": "2024-06-01", "valid_to": None},
    {"emp_id": 402, "grade": "C", "valid_from": "2024-01-01", "valid_to": None},
])
report = scd2_integrity_check(clean, "emp_id", "valid_from", "valid_to")
print(f"Clean SCD2: passed={report.passed}  entities={report.entity_count}")

# ══════════════════════════════════════════════════════════════
#  CONCEPT 1 — JINJA MACROS IN dbt
# ══════════════════════════════════════════════════════════════
# dbt models are Jinja-templated SQL files. A macro is a reusable Jinja function.
#
# A macro in macros/cents_to_dollars.sql:
#   {% macro cents_to_dollars(column_name) %}
#       ROUND({{ column_name }} / 100.0, 2)
#   {% endmacro %}
#
# Used in a model:
#   SELECT order_id, {{ cents_to_dollars('amount_cents') }} AS amount_usd
#   FROM {{ ref('stg_orders') }}

# Jinja macros can loop over lists to generate repetitive SQL.

#
# In dbt (macros/pivot_countries.sql):
#   {% macro pivot_sum(col, categories) %}
#       {% for cat in categories %}
#           SUM(CASE WHEN country = '{{ cat }}' THEN {{ col }} ELSE 0 END) AS {{ cat | lower }}_revenue

#   {% endmacro %}

def pivot_sum(col, categories, group_col):
    """Generate a PIVOT SELECT with one SUM column per category."""
    pivot_cols = ',\n        '.join(
        f"SUM(CASE WHEN {group_col} = '{cat}' THEN {col} ELSE 0 END) AS {cat.lower()}_total"
        for cat in categories
    )
    return pivot_cols

def expect_column_values_between(con, model, col, min_val, max_val, inclusive=True):
    op1, op2 = ('>=', '<=') if inclusive else ('>', '<')
    n = con.execute(f'SELECT COUNT(*) FROM {model} WHERE NOT ({col} {op1} {min_val} AND {col} {op2} {max_val})').fetchone()[0]
    return 'PASS' if n == 0 else f'FAIL ({n} out-of-range values)'

def expect_table_row_count_between(con, model, min_rows, max_rows):
    n = con.execute(f'SELECT COUNT(*) FROM {model}').fetchone()[0]
    return 'PASS' if min_rows <= n <= max_rows else f'FAIL ({n} rows, expected {min_rows}-{max_rows})'

def expect_column_mean_between(con, model, col, min_val, max_val):
    mean = con.execute(f'SELECT AVG({col}) FROM {model}').fetchone()[0]
    return 'PASS' if min_val <= mean <= max_val else f'FAIL (mean={mean:.2f}, expected {min_val}-{max_val})'

def expect_no_duplicates(con, model, cols):
    col_list = ', '.join(cols)
    n = con.execute(f'SELECT COUNT(*) FROM (SELECT {col_list} FROM {model} GROUP BY {col_list} HAVING COUNT(*) > 1)').fetchone()[0]
    return 'PASS' if n == 0 else f'FAIL ({n} duplicate combinations)'

# ══════════════════════════════════════════════════════════════
#  CONCEPT 3 — DATA CONTRACTS
# ══════════════════════════════════════════════════════════════
# A data contract is a formal agreement between the team that
# produces a dataset and the teams that consume it.
# It defines three things:
#
#   schema        — expected column names and types
#   quality_rules — assertions that must hold (not_null, range, uniqueness)
#   sla           — operational guarantees (min rows, max freshness in hours)
#
# Tools like Soda Core and Great Expectations implement data contracts
# at scale. Here we implement the same pattern with Pydantic (W2 Royal Road).
#
# Why contracts matter:
#   Without a contract, a producer can rename a column and silently break
#   every downstream dashboard and pipeline.
#   With a contract, the violation is caught at the boundary before it
#   propagates — the same principle as the Self-Healing Pipeline in W17_D7.
#
# EXAMPLE ──────────────────────────────────────────────────────

from dataclasses import dataclass, field
from typing import Any

class ContractViolationError(Exception):
    pass

@dataclass
class QualityRule:
    column:  str
    check:   str          # 'not_null' | 'between' | 'unique'
    min_val: float = None
    max_val: float = None

@dataclass
class DataContract:
    model:         str
    schema:        dict             # {col_name: expected_dtype_str}
    quality_rules: list             # list of QualityRule
    sla:           dict = field(default_factory=dict)  # {min_rows, max_freshness_hours}

def validate_contract(df, contract: DataContract):
    """Validate a DataFrame against a DataContract. Raises ContractViolationError on breach."""
    import pandas as pd
    violations = []

    # Schema check
    for col, expected_type in contract.schema.items():
        if col not in df.columns:
            violations.append(f"Missing column: {col}")

    # Quality rules
    for rule in contract.quality_rules:
        col = rule.column
        if col not in df.columns:
            continue
        if rule.check == 'not_null':
            nulls = df[col].isnull().sum()
            if nulls > 0:
                violations.append(f"{col}: not_null failed ({nulls} nulls)")
        elif rule.check == 'between':
            bad = df[(df[col] < rule.min_val) | (df[col] > rule.max_val)]
            if len(bad) > 0:
                violations.append(f"{col}: between({rule.min_val}, {rule.max_val}) failed ({len(bad)} violations)")
        elif rule.check == 'unique':
            dupes = df[col].duplicated().sum()
            if dupes > 0:
                violations.append(f"{col}: unique failed ({dupes} duplicates)")

    # SLA: min rows
    if 'min_rows' in contract.sla and len(df) < contract.sla['min_rows']:
        violations.append(f"SLA: min_rows={contract.sla['min_rows']} failed (got {len(df)})")

    if violations:
        raise ContractViolationError('\n  '.join(['Contract violations:'] + violations))
    print(f"Contract PASS: {contract.model} ({len(df)} rows)")

def downstream(dag, node, visited=None):
    """Find all models that depend (directly or indirectly) on node."""
    if visited is None:

        if node in deps and model not in visited:
            visited.add(model)
            result.append(model)
            result.extend(downstream(dag, model, visited))
    return result

# ══════════════════════════════════════════════════════════════
# The metric drift problem (why this matters):
#
#   Three BI dashboards, three definitions of "MAU":
#     - Product team counts any user who opened the app
#     - Marketing counts users who clicked a campaign link
#     - Finance counts users who generated revenue
#
#   Two executives walk into a Monday review. One says "MAU is 240K."
#   The other says "MAU is 190K." Both are reading from the same company's
#   data. The meeting derails into a 45-minute argument about definitions
#   instead of decisions. The finance deck used last quarter's cohort filter;
#   the product deck excluded inactive accounts.
#
#   This is metric drift: the same word means different things in different
#   reports. It erodes trust in data and wastes senior time.
#
# The fix: centralise every metric definition in one registry.
# Any query that asks for "MAU" reads from the same YAML config.
# If the definition changes, it changes everywhere at once.
#
# Problem: the Finance team calculates "revenue" one way, the
# Sales team calculates it another way — dashboards disagree.
#
# The dbt Semantic Layer (powered by MetricFlow) solves this by
# defining every metric ONCE in YAML on top of your mart models.
# Any BI tool, Python script, or API query reads from the same
# definition — there is one version of truth.
#
# MetricFlow concepts:
#   measure     — the raw aggregation (SUM amount, COUNT order_id)
#   dimension   — the "group by" axis (region, product, month)
#   metric      — a named measure + optional dimensions + filters
#
# In production: define in metrics.yml, query via 'mf query'.
# Here: simulated with a Python metric_registry + DuckDB.
#
# EXAMPLE ──────────────────────────────────────────────────────

import duckdb

con_sem = duckdb.connect(':memory:')
con_sem.execute('''
    CREATE TABLE fct_sales AS SELECT * FROM (VALUES
        (1, 'Widget', 'EMEA', 250.00, '2024-01'),
        (2, 'Gadget', 'AMER',  89.50, '2024-01'),
        (3, 'Widget', 'APAC', 412.00, '2024-02'),
        (4, 'Gadget', 'EMEA', 175.00, '2024-02'),
        (5, 'Widget', 'AMER', 320.00, '2024-02')
    ) AS t(sale_id, product, region, amount, month)
''')

metric_registry = {}

def define_metric(name, model, measure_col, agg, dimensions):
    """Register a metric definition."""
    metric_registry[name] = {
        'model':      model,
        'measure':    measure_col,
        'agg':        agg,           # 'SUM' | 'AVG' | 'COUNT'
        'dimensions': dimensions,    # list of valid group-by columns
    }
    print(f'Metric registered: {name}  ({agg}({measure_col}) on {model})')

def query_metric(name, group_by=None, where=None):
    """Query a registered metric, optionally grouped and filtered."""
    m = metric_registry[name]
    select  = f"{m['agg']}({m['measure']}) AS {name}"
    groupby = ''
    if group_by:
        assert group_by in m['dimensions'], f"{group_by} not a declared dimension of {name}"
        select  = f"{group_by}, {select}"
        groupby = f"GROUP BY {group_by} ORDER BY {group_by}"
    where_clause = f"WHERE {where}" if where else ''
    sql = f"SELECT {select} FROM {m['model']} {where_clause} {groupby}"
    return con_sem.execute(sql).fetchdf()

define_metric(
    name='total_revenue',
    model='fct_sales',
    measure_col='amount',
    agg='SUM',
    dimensions=['region', 'product', 'month'],
)

print()
print('total_revenue by region:')
print(query_metric('total_revenue', group_by='region').to_string(index=False))

import duckdb

def ref(m): return m
def source(s, t): return f'{s}__{t}'

def run_model(con, name, sql, materialized='view'):
    if materialized == 'table':
        con.execute(f'CREATE OR REPLACE TABLE {name} AS {sql}')
    else:
        con.execute(f'CREATE OR REPLACE VIEW {name} AS {sql}')
    rows = con.execute(f'SELECT COUNT(*) FROM {name}').fetchone()[0]
    print(f'  [{materialized:>12}]  {name}  ({rows} rows)')

def upsert(con, target, source_tbl, unique_key, cols):
    col_list = ', '.join(cols)
    tables = [r[0] for r in con.execute('SHOW TABLES').fetchall()]
    if target not in tables:
        con.execute(f'CREATE TABLE {target} AS SELECT {col_list} FROM {source_tbl}')
        mode = 'FULL LOAD'
    else:
        con.execute(f'DELETE FROM {target} WHERE {unique_key} IN (SELECT {unique_key} FROM {source_tbl})')
        con.execute(f'INSERT INTO {target} SELECT {col_list} FROM {source_tbl}')
        mode = 'UPSERT'
    n = con.execute(f'SELECT COUNT(*) FROM {target}').fetchone()[0]
    print(f'  [{mode:>12}]  {target}  ({n} rows)')

def expect_between(con, model, col, lo, hi):
    n = con.execute(f'SELECT COUNT(*) FROM {model} WHERE NOT ({col} >= {lo} AND {col} <= {hi})').fetchone()[0]
    return 'PASS' if n == 0 else f'FAIL ({n} out of range)'

def expect_no_nulls(con, model, col):
    n = con.execute(f'SELECT COUNT(*) FROM {model} WHERE {col} IS NULL').fetchone()[0]
    return 'PASS' if n == 0 else f'FAIL ({n} nulls)'
