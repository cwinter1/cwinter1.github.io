# ══════════════════════════════════════════════════════════════
#  CONCEPT 1 — WHAT dbt DOES
# ══════════════════════════════════════════════════════════════
# dbt (data build tool) sits in the TRANSFORM layer of ELT pipelines.
#
# EL is done by tools like Fivetran, Airbyte, or your own Python scripts.
# dbt does the T: it runs SQL SELECT statements inside your warehouse
# (BigQuery, Snowflake, Redshift, DuckDB) and materializes them as tables or views.
#
# Key idea: a dbt MODEL is just a .sql file with a SELECT statement.
# dbt figures out the CREATE TABLE / INSERT logic automatically.
#
# dbt also:
#   - resolves dependencies between models (runs them in the right order)
#   - runs data quality tests after each build
#   - generates documentation and a lineage DAG
#   - works with version control (git) as the source of truth
#
# SIMULATION NOTE:
#   In this curriculum we simulate dbt using DuckDB + Python.
#   Every concept shown here maps 1-to-1 to real dbt commands.
#   Real dbt commands are shown as comments.
#
# EXAMPLE ──────────────────────────────────────────────────────────
# A dbt model file looks like this (models/staging/stg_orders.sql):
#
#   select
#       order_id,
#       customer_id,
#       order_date,
#       amount
#   from {{ source('raw', 'orders') }}
#   where order_date >= '2024-01-01'
#
# dbt compiles that to:
#   CREATE VIEW staging.stg_orders AS
#   SELECT order_id, customer_id, order_date, amount
#   FROM raw.orders WHERE order_date >= '2024-01-01'
#
# In our simulation: we run the SQL directly on DuckDB.

con = duckdb.connect(':memory:')

# Create a raw table (this is what Fivetran/Airbyte would load)
con.execute('''
    CREATE TABLE raw_orders (
        order_id    INTEGER,
        customer_id INTEGER,
        order_date  DATE,
        amount      DECIMAL(10,2)
    )
''')
con.execute('''
    INSERT INTO raw_orders VALUES
        (1, 101, '2024-03-01', 250.00),
        (2, 102, '2024-03-02', 89.50),
        (3, 101, '2024-03-03', 412.00),
        (4, 103, '2023-12-15', 75.00)
''')

# This is what a dbt staging model does: select and filter raw data
stg_orders_sql = '''
    SELECT order_id, customer_id, order_date, amount
    FROM raw_orders
    WHERE order_date >= '2024-01-01'
'''

result = con.execute(stg_orders_sql).fetchdf()
print('stg_orders model output:')
print(result)

# ══════════════════════════════════════════════════════════════
#  CONCEPT 2 — dbt PROJECT FOLDER LAYOUT
# ══════════════════════════════════════════════════════════════
# When you run 'dbt init my_project', it creates:
#
#   my_project/
#   ├── dbt_project.yml    <- project config (name, version, model paths)
#   ├── profiles.yml       <- connection config (host, db, schema, credentials)
#   ├── models/            <- your SQL transformation files
#   │   ├── staging/        <- stg_* models: rename, cast, filter raw data
#   │   ├── intermediate/   <- int_* models: joins, business logic
#   │   └── mart/           <- fct_* and dim_* models: final business tables
#   ├── seeds/             <- CSV files loaded as tables (small reference data)
#   ├── tests/             <- custom SQL tests (singular tests)
#   ├── macros/            <- reusable Jinja functions
#   └── snapshots/         <- SCD Type 2 slowly changing dimension configs
#
# EXAMPLE ──────────────────────────────────────────────────────────
# A dbt_project.yml file looks like this:
dbt_project_yml = {
    'name': 'sales_pipeline',
    'version': '1.0.0',
    'profile': 'sales_pipeline',          # matches profiles.yml
    'model-paths': ['models'],
    'seed-paths': ['seeds'],
    'test-paths': ['tests'],
    'macro-paths': ['macros'],
    'models': {
        'sales_pipeline': {
            'staging': {'materialized': 'view'},       # staging = cheap views
            'intermediate': {'materialized': 'view'},
            'mart': {'materialized': 'table'}           # mart = permanent tables
        }
    }
}

for section, value in dbt_project_yml['models']['sales_pipeline'].items():
    print(f"  models/{section}/  -> materialized as {value['materialized']}")

# ══════════════════════════════════════════════════════════════
#  CONCEPT 3 — MATERIALIZATIONS
# ══════════════════════════════════════════════════════════════
# Materialization = how dbt persists a model in the warehouse.
#
#   view        -> CREATE OR REPLACE VIEW (no data stored, always fresh)
#                  Use for: staging models, lightweight transformations
#                  Cost: query runs every time the view is queried
#
#   table       -> DROP + CREATE TABLE AS SELECT (data stored physically)
#                  Use for: mart models, wide aggregations, final outputs
#                  Cost: full rebuild every dbt run
#
#   incremental -> INSERT only new/changed rows since last run
#                  Use for: large event tables (millions of rows)
#                  Cost: only processes the delta
#
#   ephemeral   -> compiled as a CTE, never persisted
#                  Use for: intermediate logic you do not want stored
#
# In dbt_project.yml or in the model file itself:
#   {{ config(materialized='table') }}
#
# EXAMPLE ──────────────────────────────────────────────────────────
scenarios = [
    ('stg_orders',       'Rename + cast raw columns, no joins'),
    ('int_orders_joined', 'Join orders with customers'),
    ('fct_daily_revenue', 'Daily revenue aggregation, queried by BI tool'),
    ('dim_customers',    'Customer dimension table, 2M rows'),
    ('util_date_spine',  'Utility CTE used inside 3 other models'),
]

recommended = {
    'stg_orders': 'view',
    'int_orders_joined': 'view',
    'fct_daily_revenue': 'table',
    'dim_customers': 'incremental',
    'util_date_spine': 'ephemeral',
}

for model, desc in scenarios:
    mat = recommended[model]
    print(f"  {model:<22} -> {mat:<12}  ({desc})")

# ══════════════════════════════════════════════════════════════
#  CONCEPT 1 — A MODEL IS A SQL SELECT
# ══════════════════════════════════════════════════════════════
# A dbt model file (models/staging/stg_orders.sql) contains ONLY a SELECT.
# No CREATE TABLE. No INSERT. dbt adds that based on the materialization config.
#

#   stg_  -> staging model (clean raw data)
#   int_  -> intermediate model (joins, business logic)

# ══════════════════════════════════════════════════════════════
#  CONCEPT 2 — ref() FOR DEPENDENCY CHAINING
# ══════════════════════════════════════════════════════════════
# ref('model_name') is how dbt models reference each other.
# It does two things:
#   1. Resolves to the correct schema.table in the warehouse at compile time
#   2. Tells dbt "this model DEPENDS ON that model" — builds the DAG
#
# Without ref(), if you hardcode a table name:
#   SELECT * FROM analytics.stg_orders          <- WRONG: breaks in different environments
# With ref():
#   SELECT * FROM {{ ref('stg_orders') }}       <- CORRECT: resolves to right env automatically
#
# EXAMPLE ──────────────────────────────────────────────────────────
# stg_bookings already exists from Concept 1.

# ══════════════════════════════════════════════════════════════
#  CONCEPT 3 — BUILDING THE LINEAGE DAG
# ══════════════════════════════════════════════════════════════
# Every ref() call creates a directed edge in the dbt DAG.
# dbt uses this to determine execution order: upstream models run first.
#
# Example DAG:
#   raw_orders  raw_customers
#       |              |
#   stg_orders   stg_customers
#       |    \          |
#       |    int_orders_customers
#       |              |
#            fct_revenue
#
# 'dbt run' executes in topological order. 'dbt run -s fct_revenue+'
# runs fct_revenue and everything downstream of it.
#
# EXAMPLE ──────────────────────────────────────────────────────────
# Simulate the DAG execution order
dag = {
    'stg_bookings':         [],
    'stg_hotels':           [],
    'int_bookings_hotels':  ['stg_bookings', 'stg_hotels'],
    'fct_revenue':          ['int_bookings_hotels'],
}

def topological_sort(dag):
    visited, order = set(), []
    def visit(node):
        if node in visited:
            return
        visited.add(node)
        for dep in dag.get(node, []):
            visit(dep)
        order.append(node)
    for node in dag:
        visit(node)
    return order

execution_order = topological_sort(dag)
print('dbt execution order:')
for i, model in enumerate(execution_order, 1):
    deps = dag[model]
    dep_str = f"depends on: {deps}" if deps else "no dependencies"
    print(f'  {i}. {model}  ({dep_str})')

# ══════════════════════════════════════════════════════════════
#  CONCEPT 1 — SOURCES VS MODELS
# ══════════════════════════════════════════════════════════════
# Sources are raw tables loaded into your warehouse by EL tools (Fivetran, Airbyte).
# You do NOT create or own them. dbt just needs to know they exist.
#
# Why not just hardcode the table name?
#   SELECT * FROM raw.orders   <- breaks if the schema is renamed, no lineage tracking
#
# With dbt sources:
#   SELECT * FROM {{ source('raw', 'orders') }}
#   -> dbt knows this model depends on raw.orders
#   -> dbt can test freshness: alert if orders data is > 24h old
#   -> lineage graph shows raw.orders -> stg_orders -> fct_revenue
#
# Sources are declared in sources.yml (or any YAML file in models/).
#
# EXAMPLE ──────────────────────────────────────────────────────────
# A sources.yml file looks like this:
sources_yml = {
    'version': 2,
    'sources': [
        {
            'name': 'raw',                       # source name used in source('raw', ...)
            'database': 'production',            # BigQuery project or warehouse db
            'schema': 'raw_data',                # schema where raw tables live
            'tables': [
                {'name': 'orders',
                 'description': 'Orders loaded by Fivetran from Shopify'},
                {'name': 'customers',
                 'description': 'Customer records from CRM'},
                {'name': 'products',
                 'description': 'Product catalog from ERP'}
            ]
        }
    ]
}

for table in sources_yml['sources'][0]['tables']:
    src_name = sources_yml['sources'][0]['name']
    print(f"  source('{src_name}', '{table['name']}')  ->  {table['description']}")

# ══════════════════════════════════════════════════════════════
#  CONCEPT 2 — source() IN A STAGING MODEL
# ══════════════════════════════════════════════════════════════
# Staging models always read from source(), never from ref().

# dbt can check when a source was last updated and alert if it is stale.

#
#   loaded_at_field: _fivetran_synced   <- timestamp column dbt checks

from datetime import datetime, timedelta

def check_source_freshness(last_loaded_at, warn_hours=12, error_hours=24):
    """Simulate dbt source freshness check."""
    age_hours = (datetime.now() - last_loaded_at).total_seconds() / 3600
    if age_hours > error_hours:
        return 'ERROR', f'Source is {age_hours:.1f}h old (limit: {error_hours}h)'
    elif age_hours > warn_hours:
        return 'WARN',  f'Source is {age_hours:.1f}h old (limit: {warn_hours}h)'
    return 'PASS', f'Source is {age_hours:.1f}h old — fresh'

# dbt test simulation helpers
def test_not_null(con, model, column):
    count = con.execute(f'SELECT COUNT(*) FROM {model} WHERE {column} IS NULL').fetchone()[0]
    return 'PASS' if count == 0 else f'FAIL ({count} nulls found)'

def test_unique(con, model, column):
    count = con.execute(f'SELECT COUNT(*) FROM (SELECT {column} FROM {model} GROUP BY {column} HAVING COUNT(*) > 1)').fetchone()[0]
    return 'PASS' if count == 0 else f'FAIL ({count} duplicate values found)'

def test_accepted_values(con, model, column, values):
    vals = ', '.join(f"'{v}'" for v in values)
    count = con.execute(f'SELECT COUNT(*) FROM {model} WHERE {column} NOT IN ({vals})').fetchone()[0]
    return 'PASS' if count == 0 else f'FAIL ({count} unexpected values found)'

def test_relationships(con, model, column, ref_model, ref_column):
    count = con.execute(f'SELECT COUNT(*) FROM {model} WHERE {column} NOT IN (SELECT {ref_column} FROM {ref_model})').fetchone()[0]
    return 'PASS' if count == 0 else f'FAIL ({count} orphaned foreign keys)'

def call_llm(system_prompt: str, user_prompt: str) -> str:
    """
    SIMULATION MODE — deterministic response for known failure patterns.
    REAL MODE (uncomment + set GROQ_API_KEY in .env):

    from openai import OpenAI
    import os
    client = OpenAI(
        api_key=os.getenv('GROQ_API_KEY'),
        base_url='https://api.groq.com/openai/v1',

    resp = client.chat.completions.create(
        model='llama3-8b-8192',
        messages=[
            {'role': 'system', 'content': system_prompt},
            {'role': 'user',   'content': user_prompt},
        ],
        temperature=0.1,
    )
    return resp.choices[0].message.content

    if 'not_null' in user_prompt and 'department' in user_prompt:

            "The department column contains 1 NULL value — a row was loaded "
            "without a department assignment. Fix: add COALESCE(department, "

            "Fix: update accepted_values in schema.yml to include the new value, "

# ══════════════════════════════════════════════════════════════
#  CONCEPT 4 — SECURITY GATE: dbt CONTRACT GATE (CIA: I)
# ══════════════════════════════════════════════════════════════
# Analogy: the building inspector who must sign off on the foundations
# before the contractor can pour the walls — no downstream work until
# the base layer passes.
#
# CIA focus: Integrity — data contracts must be enforced on every dbt
# source before downstream models run.
#
# Checks:
#   1. Every source table must have a 'freshness' key (availability contract)
#   2. Every PII column (email, phone, ssn, national_id, passport) must have
#      at least one test defined — untested PII is a compliance gap
#
# EXAMPLE ──────────────────────────────────────────────────────────
import re
from dataclasses import dataclass, field

_PII_COL_PATTERN = re.compile(r"email|phone|ssn|national_id|passport", re.I)


class MissingFreshnessError(Exception):
    pass


class PIIColumnUntestedError(Exception):
    pass


@dataclass
class ContractReport:
    passed: bool
    violations: list = field(default_factory=list)


def dbt_contract_gate(sources_config: dict) -> ContractReport:
    for table in sources_config.get("tables", []):
        name = table.get("name", "")
        if "freshness" not in table:
            raise MissingFreshnessError(f"Table '{name}' missing freshness config")
        for col in table.get("columns", []):
            col_name = col.get("name", "")
            if _PII_COL_PATTERN.search(col_name) and not col.get("tests"):
                raise PIIColumnUntestedError(
                    f"PII column '{col_name}' in '{name}' has no tests defined"
                )
    return ContractReport(passed=True)

        hotel_type,

        ROUND(SUM(avg_daily_rate), 2) AS total_revenue
    FROM {ref('stg_bookings')}
    GROUP BY 1, 2, 3

import duckdb
import os
import pandas as pd

con = duckdb.connect(':memory:')

def ref(model_name):
    return model_name

def source(schema, table):
    return f'{schema}__{table}'

def run_model(con, name, sql, materialized='view'):
    if materialized == 'table':
        con.execute(f'CREATE OR REPLACE TABLE {name} AS {sql}')
    else:
        con.execute(f'CREATE OR REPLACE VIEW {name} AS {sql}')
    rows = con.execute(f'SELECT COUNT(*) FROM {name}').fetchone()[0]
    print(f'  [{materialized:>12}]  {name}  ({rows} rows)')

def dbt_test(con, model, column, test_type, values=None, ref_model=None, ref_col=None):
    if test_type == 'not_null':
        n = con.execute(f'SELECT COUNT(*) FROM {model} WHERE {column} IS NULL').fetchone()[0]
        return 'PASS' if n == 0 else f'FAIL ({n} nulls)'
    elif test_type == 'unique':
        n = con.execute(f'SELECT COUNT(*) FROM (SELECT {column} FROM {model} GROUP BY {column} HAVING COUNT(*)>1)').fetchone()[0]
        return 'PASS' if n == 0 else f'FAIL ({n} dupes)'
    elif test_type == 'accepted_values':
        vals = ', '.join(f"'{v}'" for v in values)
        n = con.execute(f'SELECT COUNT(*) FROM {model} WHERE {column} NOT IN ({vals})').fetchone()[0]
        return 'PASS' if n == 0 else f'FAIL ({n} bad values)'
    return 'UNKNOWN'
