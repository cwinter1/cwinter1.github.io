══════════════════════════════════════════════════════════════
 CONCEPT 1 — ETL ARCHITECTURE: EXTRACT, TRANSFORM, LOAD AS A PATTERN
══════════════════════════════════════════════════════════════
ETL stands for Extract, Transform, Load -- the three stages of moving data
from a source system into a destination like a data warehouse or database.

EXTRACT:  Pull raw data from source systems (CSV, DB, API, Excel)
TRANSFORM: Clean, validate, enrich, and reshape the data
LOAD:     Write the transformed data to a destination

WHY STRUCTURE CODE THIS WAY?
  - Each stage is testable in isolation
  - Failures are easier to pinpoint (did extract fail? or transform?)
  - Stages can be swapped independently (change the source without touching transform)
  - Makes pipelines readable and maintainable

A simple ETL pipeline in Python looks like this:

  def extract(source):    ...  return raw_df
  def transform(raw_df):  ...  return clean_df
  def load(clean_df, dest): ...

  def run_pipeline():
      raw    = extract("sales_2024.csv")
      clean  = transform(raw)
      load(clean, "warehouse.db")

 CONCEPT 2 — BUILDING AN EXTRACT FUNCTION (FROM CSV FILE)
══════════════════════════════════════════════════════════════
The extract function is responsible for:
  - Reading raw data from the source
  - Returning it as a pandas DataFrame (unchanged)
  - Handling read errors (file not found, network failure, etc.)
  - NOT transforming anything -- extraction is purely about reading

def extract_from_csv_string(csv_string, source_name="inline_csv"):
    """
    Read CSV data from a string (simulates reading from a file or API).
    Returns (DataFrame, metadata_dict).
    """
    print(f"\n[EXTRACT] Reading from: {source_name}")
    try:
        df = pd.read_csv(io.StringIO(csv_string))
        rows = len(df)
        cols = len(df.columns)
        print(f"[EXTRACT] Read {rows} rows x {cols} columns")
        return df, {"rows": rows, "columns": cols, "source": source_name, "status": "ok"}
    except Exception as e:
        print(f"[EXTRACT] ERROR: {e}")
        return pd.DataFrame(), {"rows": 0, "columns": 0, "source": source_name,
                                 "status": "error", "error_message": str(e)}

def extract_from_file(filepath):
    """
    Read CSV data from a real file path.
    Returns (DataFrame, metadata_dict).
    """
    print(f"\n[EXTRACT] Reading file: {filepath}")
    try:
        df = pd.read_csv(filepath)
        print(f"[EXTRACT] Read {len(df)} rows from {os.path.basename(filepath)}")

        print(f"[EXTRACT] ERROR: File not found: {filepath}")

 CONCEPT 3 — TRANSFORM AND LOAD FUNCTIONS, CHAINING INTO A PIPELINE
══════════════════════════════════════════════════════════════
The transform function:
  - Cleans data (handle nulls, fix types, remove invalid rows)
  - Validates data (check ranges, required fields)
  - Enriches data (add calculated columns, lookups)
  - Returns a clean DataFrame

The load function:
  - Writes the clean DataFrame to a destination
  - Handles the "if_exists" logic (replace vs append)
  - Returns a count of rows written

EXAMPLE ──────────────────────────────────────────────────────
def transform_sales(df):
    """
    Clean and enrich the raw sales DataFrame.
    Returns (clean_df, metadata_dict).
    """
    print("\n[TRANSFORM] Starting transformation...")
    errors = []
    df_clean = df.copy()

    # Fix column names: strip whitespace and lowercase
    df_clean.columns = [c.strip().lower() for c in df_clean.columns]

    # Parse dates -- invalid dates become NaT
    df_clean["sale_date"] = pd.to_datetime(df_clean["sale_date"], errors="coerce")

    # Drop rows with null sale_date or null product (required fields)
    before = len(df_clean)
    df_clean = df_clean.dropna(subset=["sale_date", "product"])
    dropped = before - len(df_clean)
    if dropped > 0:
        errors.append(f"Dropped {dropped} rows with missing sale_date or product")
        print(f"[TRANSFORM] Dropped {dropped} invalid rows")

    # Convert numeric columns, coerce errors to NaN
    df_clean["quantity"]     = pd.to_numeric(df_clean["quantity"],   errors="coerce").fillna(0).astype(int)
    df_clean["unit_price"]   = pd.to_numeric(df_clean["unit_price"], errors="coerce").fillna(0.0)
    df_clean["discount_pct"] = pd.to_numeric(df_clean["discount_pct"], errors="coerce").fillna(0.0)

    # Remove rows with invalid data
    df_clean = df_clean[df_clean["quantity"] > 0]
    df_clean = df_clean[df_clean["unit_price"] > 0]

    # Add calculated columns
    df_clean["discount_amount"] = (df_clean["unit_price"] * df_clean["discount_pct"] / 100).round(2)
    df_clean["final_price"]     = (df_clean["unit_price"] - df_clean["discount_amount"]).round(2)
    df_clean["line_revenue"]    = (df_clean["quantity"] * df_clean["final_price"]).round(2)

    # Standardize region
    df_clean["region"] = df_clean["region"].str.strip().str.title()

    rows_out = len(df_clean)
    print(f"[TRANSFORM] Output: {rows_out} clean rows")
    return df_clean, {"rows_in": before, "rows_out": rows_out, "errors": errors, "status": "ok"}

def load_to_csv(df, output_path):
    """Write the DataFrame to a CSV file. Returns rows written."""
    print(f"\n[LOAD] Writing {len(df)} rows to: {output_path}")
    df.to_csv(output_path, index=False)
    print(f"[LOAD] Complete.")
    return len(df)

def run_pipeline(csv_data, output_path):
    """Run the full ETL pipeline and return a summary dict."""
    meta = {"pipeline": "daily_sales_etl", "errors": []}

    # Extract
    raw, extract_info = extract_from_csv_string(csv_data, "daily_sales_feed")
    meta["rows_extracted"] = extract_info["rows"]

    if extract_info["status"] != "ok":
        meta["status"] = "failed_at_extract"
        return meta

    # Transform
    clean, transform_info = transform_sales(raw)
    meta["rows_transformed"] = transform_info["rows_out"]
    meta["errors"].extend(transform_info["errors"])

    # Load
    rows_loaded = load_to_csv(clean, output_path)
    meta["rows_loaded"] = rows_loaded
    meta["status"] = "success"

    return meta

 CONCEPT 1 — EXTRACT FROM MULTIPLE CSV FILES IN A FOLDER
══════════════════════════════════════════════════════════════
In production, data often arrives as daily or monthly files in a folder.
Pattern:  sales_2024_01.csv, sales_2024_02.csv, sales_2024_03.csv, ...
We need to read ALL matching files and combine them into one DataFrame.

TOOLS:
  glob.glob(pattern)  -- returns a list of file paths matching a pattern
  pd.concat([df1, df2, ...])  -- stack DataFrames vertically
  pd.read_csv(path)  -- read one file at a time

GLOB PATTERNS:
  "*.csv"             -- all CSV files in the current folder
  "data/*.csv"        -- all CSVs in the 'data' subfolder
  "sales_2024_*.csv"  -- files matching the naming convention

def extract_from_folder(folder_path, pattern="*.csv"):
    """
    Read all CSV files matching pattern in folder_path.
    Returns (combined_DataFrame, list_of_files_read).
    """
    search_path = os.path.join(folder_path, pattern)
    file_paths = sorted(glob.glob(search_path))

    if not file_paths:
        print(f"[EXTRACT] No files matching '{pattern}' in {folder_path}")
        return pd.DataFrame(), []

    print(f"[EXTRACT] Found {len(file_paths)} file(s):")
    dfs = []
    for path in file_paths:
        try:
            df = pd.read_csv(path)
            df["_source_file"] = os.path.basename(path)   # track which file each row came from
            dfs.append(df)
            print(f"  {os.path.basename(path)}: {len(df)} rows")
        except Exception as e:
            print(f"  ERROR reading {path}: {e}")

    combined = pd.concat(dfs, ignore_index=True)
    print(f"[EXTRACT] Combined: {len(combined)} rows total")
    return combined, file_paths

 CONCEPT 2 — EXTRACT FROM A DATABASE (sqlite3 + pd.read_sql)
══════════════════════════════════════════════════════════════
Extracting from a database involves:
  1. Connecting to the database
  2. Writing a SELECT query
  3. Using pd.read_sql() to load results into a DataFrame
  4. Closing the connection

When extracting large tables:
  - Use WHERE to filter (only load data you actually need)
  - Use SELECT specific_columns (avoid SELECT *)
  - Use LIMIT or WHERE date > 'last_run_date' for incremental extraction
  - Use chunksize= in pd.read_sql() for very large tables

def extract_from_database(connection, query, params=None):
    """
    Execute a SQL query and return the result as a DataFrame.
    """
    print(f"\n[EXTRACT DB] Running query...")
    try:
        if params:
            df = pd.read_sql(query, connection, params=params)
        else:
            df = pd.read_sql(query, connection)

        return df, {"rows": len(df), "status": "ok"}
    except Exception as e:
        print(f"[EXTRACT DB] ERROR: {e}")
        return pd.DataFrame(), {"rows": 0, "status": "error", "error": str(e)}

══════════════════════════════════════════════════════════════
Extracting from an API follows a consistent pattern:
  1. Make a GET request to the endpoint
  2. Parse the JSON response
  3. Convert to a DataFrame
  4. Handle pagination if the API returns data in pages

def extract_from_api(url, params=None, timeout=10):
    """
    Fetch JSON data from a REST API and return as a DataFrame.
    Returns (DataFrame, metadata_dict).
    """
    try:
        import requests
        print(f"\n[EXTRACT API] GET {url}")
        response = requests.get(url, params=params, timeout=timeout)
        response.raise_for_status()
        data = response.json()
        # Handle both list responses and nested responses
        if isinstance(data, list):
            df = pd.DataFrame(data)
        elif isinstance(data, dict):
            df = pd.DataFrame([data])
        else:
            df = pd.DataFrame()
        print(f"[EXTRACT API] {len(df)} records returned")
        return df, {"rows": len(df), "status": "ok", "url": url}
    except ImportError:
        print("[EXTRACT API] requests not installed. Run: pip install requests")

# ══════════════════════════════════════════════════════════════

import json

# Simulate a Kafka topic as a deque (first-in first-out queue)
_topics = {}  # topic_name -> deque of messages

def kafka_produce(topic, message: dict):
    """Append a JSON message to a topic."""
    if topic not in _topics:
        _topics[topic] = deque()
    _topics[topic].append(json.dumps(message))

def kafka_consume(topic, group_id, max_msgs=None):
    """
    Pop messages from a topic and yield (offset, message_dict).
    Tracks offset per consumer group in a simple dict.
    """
    if topic not in _topics:
        return
    _offsets = {}
    key = f'{group_id}::{topic}'
    count = 0
    while _topics[topic]:
        if max_msgs and count >= max_msgs:
            break
        raw = _topics[topic].popleft()
        _offsets[key] = _offsets.get(key, -1) + 1
        yield _offsets[key], json.loads(raw)
        count += 1

 CONCEPT 1 — STANDARDIZING COLUMN NAMES
══════════════════════════════════════════════════════════════
Inconsistent column names cause errors and make code hard to write.
The standard in data engineering is:
  - All lowercase
  - Words separated by underscores
  - No leading or trailing spaces
  - No special characters

This is called "snake_case" naming.

TECHNIQUES:
  df.columns = [c.strip().lower().replace(" ", "_") for c in df.columns]
  df.rename(columns={"Old Name": "new_name"}) -- rename specific columns
  df.columns = df.columns.str.strip().str.lower().str.replace(" ", "_")

def standardize_column_names(df):
    """
    Strip whitespace, lowercase, and replace spaces with underscores.
    Returns a new DataFrame with clean column names.
    """
    new_cols = []
    for col in df.columns:
        clean = col.strip()                          # remove leading/trailing spaces
        clean = clean.lower()                        # lowercase
        clean = clean.replace(" ", "_")              # spaces to underscores
        clean = clean.replace("?", "")               # remove punctuation
        clean = clean.replace("(", "").replace(")", "")
        new_cols.append(clean)
    df_clean = df.copy()
    df_clean.columns = new_cols
    return df_clean

 CONCEPT 2 — DATA VALIDATION
══════════════════════════════════════════════════════════════
Data validation checks that the data meets expected quality standards BEFORE
it enters a database or report. Catching bad data early prevents downstream errors.

COMMON VALIDATION CHECKS:
  1. Null check:       required columns have no NaN values
  2. Range check:      numeric values are within acceptable bounds
  3. Type check:       columns have the right data type
  4. Domain check:     categorical values come from an allowed set
  5. Uniqueness check: key columns have no duplicates
  6. Format check:     strings match expected patterns (dates, IDs)

A validation function should:
  - Return True/False for each rule
  - Collect ALL failures before stopping (so you fix all at once)
  - Log which rows violated each rule

EXAMPLE ──────────────────────────────────────────────────────
def validate_employees(df):
    """
    Run validation checks on the employees DataFrame.
    Returns a validation report dict.
    """
    report = {"checks": [], "total_failures": 0, "is_valid": True}

    def add_check(rule_name, failing_count, detail=""):
        status = "PASS" if failing_count == 0 else "FAIL"
        report["checks"].append({
            "rule": rule_name,
            "status": status,
            "failures": failing_count,
            "detail": detail,
        })
        if failing_count > 0:
            report["total_failures"] += failing_count
            report["is_valid"] = False

    # Check 1: Required columns not null
    required_cols = ["employee_id", "full_name", "department_code", "salary"]
    for col in required_cols:
        if col in df.columns:
            nulls = df[col].isnull().sum() + (df[col] == "").sum()
            add_check(f"no_null_{col}", nulls, f"{nulls} null/empty values")

    # Check 2: Salary range
    if "salary" in df.columns:
        sal = pd.to_numeric(df["salary"], errors="coerce")
        below_min = (sal < 20000).sum()
        above_max = (sal > 500000).sum()
        add_check("salary_min_20000", below_min,
                  df.loc[sal < 20000, "employee_id"].tolist() if below_min else "")
        add_check("salary_max_500000", above_max,
                  df.loc[sal > 500000, "employee_id"].tolist() if above_max else "")

    # Check 3: Valid department codes
    valid_depts = {"ENG", "SLS", "FIN", "OPS", "HR"}
    if "department_code" in df.columns:
        invalid_depts = df[~df["department_code"].isin(valid_depts) & df["department_code"].notna()]
        add_check("valid_department_code", len(invalid_depts),
                  invalid_depts["employee_id"].tolist() if len(invalid_depts) else "")

    # Check 4: Performance score 0-5
    if "performance_score" in df.columns:
        score = pd.to_numeric(df["performance_score"], errors="coerce")
        out_of_range = ((score < 0) | (score > 5)).sum()
        add_check("performance_score_0_to_5", out_of_range)

    return report

 CONCEPT 3 — BUSINESS RULES TRANSFORMATION
══════════════════════════════════════════════════════════════
Business rules transform raw data into business-meaningful values.
These are domain-specific calculations and mappings your organization defines.

COMMON PATTERNS:
  1. Derived columns: calculate new columns from existing ones
     e.g. revenue = quantity * unit_price
     e.g. tenure_years = (today - hire_date).days / 365

  2. Category mapping: convert codes to human-readable labels
     e.g. dept_code "ENG" -> "Engineering"

  3. Bucketing/binning: group continuous values into categories
     e.g. salary < 60k = "Junior", 60k-90k = "Mid", 90k+ = "Senior"

  4. Flag creation: create boolean indicators
     e.g. is_high_performer = performance_score >= 4.0

EXAMPLE ──────────────────────────────────────────────────────
def apply_business_rules(df):
    """
    Apply business logic transformations to the employees DataFrame.
    Returns transformed DataFrame.
    """
    df = df.copy()

    # 1. Parse and standardize the start_date
    df["start_date"] = pd.to_datetime(df["start_date"], errors="coerce")

    # 2. Calculate tenure in years (rounded to 1 decimal)
    today = pd.Timestamp("2024-12-31")   # use fixed date for reproducibility
    df["tenure_years"] = ((today - df["start_date"]).dt.days / 365.25).round(1)
    df["tenure_years"] = df["tenure_years"].clip(lower=0)   # no negative tenure

    # 3. Standardize full_name
    df["full_name"] = df["full_name"].str.strip().str.title()

    # 4. Standardize status
    df["status"] = df["status"].str.strip().str.lower().str.capitalize()

    # 5. Map department code to full name
    dept_map = {
        "ENG": "Engineering",
        "SLS": "Sales",
        "FIN": "Finance",
        "OPS": "Operations",
        "HR":  "Human Resources",
    }
    df["department"] = df["department_code"].map(dept_map).fillna("Unknown")

    # 6. Salary band bucketing
    salary_col = pd.to_numeric(df["salary"], errors="coerce")
    df["salary_band"] = pd.cut(
        salary_col,
        bins=[0, 60000, 90000, 120000, float("inf")],
        labels=["Junior", "Mid", "Senior", "Executive"],
        right=True,
    )

    # 7. Boolean flags
    df["is_manager"]      = df["manager"].str.lower().str.strip().isin(["yes"]) if "manager" in df.columns else False
    df["is_high_performer"] = pd.to_numeric(df["performance_score"], errors="coerce") >= 4.0
    df["is_active"]       = df["status"].str.lower() == "active"

    # 8. Performance tier
    score = pd.to_numeric(df["performance_score"], errors="coerce")
    df["performance_tier"] = pd.cut(
        score,
        bins=[0, 2.5, 3.5, 4.5, 5.0],
        labels=["Needs Improvement", "Meets Expectations", "Exceeds", "Outstanding"],
        right=True,
    )

    return df

# ══════════════════════════════════════════════════════════════
#  CONCEPT 4 -- SCHEMA DRIFT DETECTION
# ══════════════════════════════════════════════════════════════
#
#  Schema drift happens when an upstream data source changes its
#  column names or structure without warning. A pipeline that
#  expects 'user_id' will crash silently if it receives 'customer_id'.
#
#  Production pipelines detect drift before processing:
#    expected_cols = {'user_id', 'email', 'signup_date', 'country'}
#    actual_cols   = set(df.columns)
#
#    added   = actual_cols - expected_cols   # new columns in source
#    removed = expected_cols - actual_cols   # columns that disappeared
#
#  A robust pipeline:
#    1. Compares column sets on every run
#    2. Logs drift as a warning (not a crash) -- missing cols filled with None
#    3. Flags potential renames (column removed + column added -> likely renamed)
#    4. Sends a Slack alert or writes to an audit table
#
#  EXAMPLE ----------------------------------------------------------
import pandas as pd

def detect_schema_drift(df: pd.DataFrame, expected_cols: set) -> dict:
    """
    Compare actual DataFrame columns against the expected schema.
    Returns a drift report dict.
    """
    actual_cols = set(df.columns)
    added       = actual_cols - expected_cols
    removed     = expected_cols - actual_cols
    report = {
        'status':      'ok' if not added and not removed else 'drift_detected',
        'added':       sorted(added),
        'removed':     sorted(removed),
        'matched':     sorted(actual_cols & expected_cols),
        'rename_hint': sorted(added) if len(added) == len(removed) else [],
    }
    return report

 CONCEPT 1 — LOADING TO CSV (FULL AND INCREMENTAL/APPEND)
══════════════════════════════════════════════════════════════

FULL LOAD:
  Write the entire DataFrame to a new file (or overwrite existing).
  Used when you want a fresh, complete snapshot each time.
  df.to_csv("output.csv", index=False)

INCREMENTAL/APPEND LOAD:
  Add new rows to an existing file without rewriting it.
  Used for logs, history tables, and cumulative exports.
  df.to_csv("output.csv", mode="a", header=False, index=False)

KEY PARAMETERS:
  index=False          -- do not write the row number as a column
  mode="w"             -- write (overwrite) [default]
  mode="a"             -- append to existing file
  header=True/False    -- write column names as header row
  encoding="utf-8"     -- consistent text encoding

EXAMPLE ──────────────────────────────────────────────────────
def load_to_csv_full(df, output_path):
    """
    Full refresh: overwrite the CSV with the entire DataFrame.
    Returns the number of rows written.
    """
    df.to_csv(output_path, index=False, encoding="utf-8")
    print(f"[LOAD CSV] Full load: {len(df)} rows -> {os.path.basename(output_path)}")
    return len(df)

def load_to_csv_append(df, output_path):
    """
    Incremental append: add new rows to an existing CSV.
    If the file does not exist, creates it with a header.
    Returns the number of rows appended.
    """
    file_exists = os.path.exists(output_path)
    df.to_csv(
        output_path,
        mode="a",
        header=not file_exists,  # write header only if file is new
        index=False,
        encoding="utf-8",
    )
    print(f"[LOAD CSV] Appended: {len(df)} rows -> {os.path.basename(output_path)}")
    return len(df)

def verify_csv_load(source_df, output_path):
    """
    Verify a CSV load was successful.
    Returns a dict with verification results.
    """
    result = {
        "output_path": output_path,
        "expected_rows": len(source_df),
        "actual_rows": None,
        "row_count_match": False,
        "sample_match": False,
        "status": "unknown",
    }

    try:
        df_loaded = pd.read_csv(output_path)
        result["actual_rows"] = len(df_loaded)
        result["row_count_match"] = (result["actual_rows"] == result["expected_rows"])

        # Sample check: compare first and last row values
        if len(df_loaded) > 0 and len(source_df) > 0:
            first_match = (source_df.iloc[0].tolist() == df_loaded.iloc[0].tolist())
            last_match  = (source_df.iloc[-1].tolist() == df_loaded.iloc[-1].tolist())
            result["sample_match"] = first_match and last_match

        result["status"] = "pass" if (result["row_count_match"] and result["sample_match"]) else "fail"
    except Exception as e:
        result["status"] = "error"
        result["error"] = str(e)

    return result

 CONCEPT 4 — SMART SCHEMA AGENT: STAGE 2 — AUDIT LOG
══════════════════════════════════════════════════════════════
 CUMULATIVE PROJECT — Stage 2 builds on Stage 1 (W6).
 The agent now records every query it executes to an audit_log table.

 Why this matters: production systems must be auditable. When something goes
 wrong at 2am, the audit log tells you what ran, when, and whether it failed.
 This is also a compliance requirement in regulated industries.

 New this week:
   - _init_audit_log(): creates the audit_log table on startup
   - execute_query(sql, params): runs the query and writes a log entry
   - get_audit_history(limit): reads back the last N log entries

 The audit_log is just another table in the same SQLite database.

 EXAMPLE ──────────────────────────────────────────────────────
import sqlite3
from datetime import datetime


class SmartSchemaAgent:
    """
    A governed database agent. Grows each week (W6-W11).
    Stage 2 — adds audit logging for every executed query.
    """

    def __init__(self, conn: sqlite3.Connection):
        self.conn = conn
        self.cur  = conn.cursor()
        self._init_audit_log()

    def _init_audit_log(self):
        self.cur.execute('''
            CREATE TABLE IF NOT EXISTS audit_log (
                log_id    INTEGER PRIMARY KEY AUTOINCREMENT,
                timestamp TEXT    NOT NULL,
                sql_text  TEXT    NOT NULL,
                row_count INTEGER,
                status    TEXT    NOT NULL
            )
        ''')
        self.conn.commit()

    def _write_log(self, timestamp, sql, row_count, status):
        self.cur.execute(
            "INSERT INTO audit_log (timestamp, sql_text, row_count, status) VALUES (?,?,?,?)",
            (timestamp, sql[:200], row_count, status),
        )
        self.conn.commit()

    def execute_query(self, sql: str, params: tuple = ()) -> list:
        """Execute any SQL and record the outcome in audit_log."""
        ts = datetime.now().strftime("%Y-%m-%d %H:%M:%S")
        try:
            self.cur.execute(sql, params)
            rows = self.cur.fetchall()
            self.conn.commit()
            self._write_log(ts, sql, len(rows), "OK")
            return rows
        except Exception as e:
            self._write_log(ts, sql, 0, f"ERROR: {e}")
            raise

    def get_audit_history(self, limit: int = 10) -> list:
        """Return the last N audit log entries, newest first."""
        self.cur.execute(
            "SELECT log_id, timestamp, sql_text, row_count, status "
            "FROM audit_log ORDER BY log_id DESC LIMIT ?",
            (limit,),
        )
        return self.cur.fetchall()

══════════════════════════════════════════════════════════════
 CONCEPT 5 — SECURITY GATE: PIPELINE INTEGRITY  (CIA: I)
══════════════════════════════════════════════════════════════

 Analogy: the courier's tamper-evident seal — if the package was opened
 in transit, the seal breaks and the recipient knows before signing for it.

 Problem: a DataFrame travels through multiple pipeline stages (extract →
 transform → load). Any stage could corrupt, truncate, or silently alter rows.
 Without a fingerprint, a 1-row change is invisible until a downstream report
 shows wrong numbers two weeks later.

 Pattern:
   1. At extraction time: compute sha256 hash of df.to_csv(index=False)
   2. Store the hash in the audit_log alongside the batch metadata
   3. At load time: recompute hash of the DataFrame about to be written
   4. Compare: if they differ, raise TamperDetectedError and halt the load

 What changes the hash:
   - Any cell value changed (even by 1 cent in a salary column)
   - Any row added or deleted
   - Column order change (to_csv is order-sensitive)

 What does NOT trigger the hash:
   - Re-running the same transformation with the same input

 Libraries: hashlib.sha256 (stdlib, no install)

EXAMPLE ──────────────────────────────────────────────────────
import hashlib
import pandas as pd


class TamperDetectedError(Exception):
    pass


def compute_fingerprint(df: pd.DataFrame) -> str:
    payload = df.to_csv(index=False).encode("utf-8")
    return hashlib.sha256(payload).hexdigest()


def verify_pipeline_integrity(df_at_extract: pd.DataFrame, df_at_load: pd.DataFrame) -> None:
    expected = compute_fingerprint(df_at_extract)
    actual   = compute_fingerprint(df_at_load)
    if expected != actual:
        raise TamperDetectedError(
            f"Fingerprint mismatch\n  expected: {expected}\n  actual:   {actual}"
        )
    print(f"Integrity OK  sha256={expected[:16]}...")


employees = pd.DataFrame({
    "employee_id": [1, 2, 3],
    "name":        ["Alice", "Bob", "Carol"],
    "salary":      [95000, 72000, 81000],
})

fp = compute_fingerprint(employees)
print(f"Fingerprint at extract: {fp[:16]}...")

# Clean pass
verify_pipeline_integrity(employees, employees.copy())

# Tampered load

tampered.loc[1, "salary"] = 9999
try:
    verify_pipeline_integrity(employees, tampered)
except TamperDetectedError as e:

 CONCEPT 1 — LOGGING WITH PYTHON'S logging MODULE
══════════════════════════════════════════════════════════════

print() is fine for quick debugging, but for production pipelines you need:
  - Timestamps on every message
  - Severity levels (DEBUG, INFO, WARNING, ERROR, CRITICAL)
  - Log output to both the console AND a file simultaneously
  - Ability to filter by level (suppress DEBUG in production)

LOGGING LEVELS (from lowest to highest severity):
  DEBUG    -- detailed diagnostic information (development only)
  INFO     -- general operational messages (job started, rows read)
  WARNING  -- something unexpected but not a failure
  ERROR    -- a step failed but the pipeline can continue
  CRITICAL -- the pipeline cannot continue

SETUP PATTERN:
  logging.basicConfig(...)  -- one-time configuration
  logger = logging.getLogger("pipeline_name")
  logger.info("message")
  logger.error("message", exc_info=True)  -- includes stack trace

EXAMPLE ──────────────────────────────────────────────────────
def setup_logger(pipeline_name, log_dir=None, level=logging.INFO):
    """
    Create a logger with console and optional file handlers.
    Returns a configured logger.
    """
    logger = logging.getLogger(pipeline_name)
    logger.setLevel(level)

    # Prevent adding duplicate handlers if function is called multiple times
    if logger.handlers:
        logger.handlers.clear()

    # Formatter: include timestamp, level, logger name, and message
    formatter = logging.Formatter(
        fmt="%(asctime)s | %(levelname)-8s | %(name)s | %(message)s",
        datefmt="%Y-%m-%d %H:%M:%S",
    )

    # Console handler — always add this
    console_handler = logging.StreamHandler()
    console_handler.setLevel(level)
    console_handler.setFormatter(formatter)
    logger.addHandler(console_handler)

    # File handler — add only if log_dir is specified
    if log_dir:
        os.makedirs(log_dir, exist_ok=True)
        log_file = os.path.join(log_dir, f"{pipeline_name}.log")
        file_handler = logging.FileHandler(log_file, encoding="utf-8")
        file_handler.setLevel(level)
        file_handler.setFormatter(formatter)
        logger.addHandler(file_handler)
        logger.info(f"Logging to file: {log_file}")

    return logger

══════════════════════════════════════════════════════════════

A production pipeline must:
  1. Catch errors at each step and log them (not crash silently)

  - Use finally to run cleanup code (close connections, write final log entry)
  - Use raise to re-raise critical errors that must stop everything

def extract_step(csv_data, pipeline_logger):
    """Extract step with error handling."""
    pipeline_logger.info("[EXTRACT] Starting extraction")
    try:
        df = pd.read_csv(io.StringIO(csv_data))
        pipeline_logger.info(f"[EXTRACT] Read {len(df)} rows")

        raise   # re-raise to signal abort to the caller

def transform_step(df, pipeline_logger):
    """Transform step with error handling."""
    pipeline_logger.info("[TRANSFORM] Starting transformation")
    errors = []
    try:
        df["qty"]   = pd.to_numeric(df["qty"],   errors="coerce")
        df["price"] = pd.to_numeric(df["price"], errors="coerce")
        before = len(df)
        df = df.dropna(subset=["qty", "price"])
        dropped = before - len(df)
        if dropped > 0:
            pipeline_logger.warning(f"[TRANSFORM] Dropped {dropped} invalid rows")
            errors.append(f"{dropped} rows dropped")
        df["revenue"] = (df["qty"] * df["price"]).round(2)
        pipeline_logger.info(f"[TRANSFORM] Output: {len(df)} clean rows")
        return df
    except Exception as e:
        pipeline_logger.error(f"[TRANSFORM] Failed: {e}")
        raise

def load_step(df, output_path, pipeline_logger):
    """Load step with error handling."""
    pipeline_logger.info(f"[LOAD] Writing {len(df)} rows to {os.path.basename(output_path)}")
    try:
        df.to_csv(output_path, index=False)
        pipeline_logger.info("[LOAD] Complete")
        return len(df)
    except Exception as e:
        pipeline_logger.error(f"[LOAD] Failed: {e}")

 CONCEPT 1 — MOCKING EXTERNAL DEPENDENCIES
══════════════════════════════════════════════════════════════
══════════════════════════════════════════════════════════════

 A mock replaces a real object with a fake one that you control.
 Use mocks to isolate the code under test from external systems:
 APIs, databases, file systems, clocks.

 unittest.mock key tools:
   patch('module.name')       — temporarily replace an object
   MagicMock()                — a flexible fake object
   mock.return_value = x      — control what the fake returns
   mock.side_effect = [a,b]   — return different values on successive calls
   mock.assert_called_once_with(arg) — verify it was called correctly

 Rule: mock at the boundary where your code calls out to the world.
 Never mock internal functions — test those directly.

EXAMPLE ──────────────────────────────────────────────────────

 CONCEPT 2 — INTEGRATION TESTS
══════════════════════════════════════════════════════════════
══════════════════════════════════════════════════════════════

 A unit test isolates one function and mocks everything else.
 An integration test runs multiple components together against
 a real (or realistic) version of the external system.

 For data pipelines, a good integration test:
   1. Seeds a real SQLite (or test) database with known data
   2. Runs the full pipeline function (extract → transform → load)
   3. Queries the output table and asserts the result is correct
   4. Tears down (cleans up) the database after the test

 Pattern:
   def test_full_pipeline():
       conn = setup_test_db(seed_data)
       run_pipeline(conn)
       result = conn.execute('SELECT ...')
       assert result == expected
       conn.close()

EXAMPLE ──────────────────────────────────────────────────────

#  CONCEPT 1 — KAFKA FUNDAMENTALS: TOPICS, PRODUCERS, CONSUMERS
# ══════════════════════════════════════════════════════════════
#
#  Kafka is a distributed event log, not a message queue.
#  Key difference: messages are NOT deleted after being read —
#  they persist and any consumer can re-read from any offset.
#
#  Core concepts
#  ─────────────
#  TOPIC      a named stream of events  (e.g. "orders", "pageviews")
#  PARTITION  a topic is split into N partitions for parallelism
#  OFFSET     position of a message inside a partition (0, 1, 2 ...)
#  PRODUCER   writes events to a topic
#  CONSUMER   reads events from a topic, tracks its own offset
#  CONSUMER GROUP  multiple consumers share work across partitions
#                  each partition is read by exactly one member
#
#  Event anatomy
#  ─────────────
#  key    (optional) determines which partition the event goes to
#  value  the payload — typically JSON
#  timestamp  set automatically by the broker
#
#  Why use Kafka?
#  ──────────────
#  ETL pipelines decouple producers from consumers:
#    API → [Kafka topic] → ETL consumer → database
#  If the database is down, events pile up safely in Kafka.
#  When it recovers, the consumer replays from its last offset.
#

class KafkaBroker:
    """In-memory broker. Replace with a real broker for REAL MODE."""
    def __init__(self):
        self._topics: dict[str, list[dict]] = defaultdict(list)
        self._offsets: dict[tuple, int] = {}   # (group_id, topic) -> next offset

    def _produce(self, topic: str, key: str, value: str):
        self._topics[topic].append({"key": key, "value": value,
                                    "offset": len(self._topics[topic])})

    def _poll(self, group_id: str, topic: str) -> dict | None:
        key = (group_id, topic)
        offset = self._offsets.get(key, 0)
        messages = self._topics.get(topic, [])
        if offset < len(messages):
            msg = messages[offset]
            self._offsets[key] = offset + 1
            return msg
        return None


class KafkaProducer:
    def __init__(self, broker: KafkaBroker, config: dict = None):
        self._broker = broker

    def produce(self, topic: str, key: str, value: str):
        self._broker._produce(topic, key, value)

    def flush(self):
        pass   # no-op in simulation; real confluent-kafka flushes the buffer


class KafkaConsumer:
    def __init__(self, broker: KafkaBroker, group_id: str):
        self._broker = broker
        self._group_id = group_id
        self._topics: list[str] = []

    def subscribe(self, topics: list[str]):
        self._topics = topics

    def poll(self, timeout: float = 1.0) -> dict | None:
        for topic in self._topics:
            msg = self._broker._poll(self._group_id, topic)
            if msg:
                return msg
        return None

# ══════════════════════════════════════════════════════════════
#
#  In an ETL pipeline the producer is the extraction layer.
#  Instead of writing directly to a database, the extractor
#  writes events to a Kafka topic. This decouples extraction
#  from transformation and loading.
#
#  Pattern
#  ───────
#    for record in source:              read from CSV / API / DB
#        producer.produce(topic,        write to Kafka
#                         key=record[pk],
#                         value=json.dumps(record))
#    producer.flush()                   wait for all acks
#
#  Why a key?
#  ──────────
#  The key determines which partition the event lands in.
#  Use the primary key of the entity (order_id, user_id).
#  All events for the same key land in the same partition →
#  guaranteed ordering per entity.
#
#  Schema discipline
#  ─────────────────
#  Always include a schema_version field in the event payload.
#  When you change the schema, bump the version — consumers
#  can detect and handle both old and new formats.
#
# EXAMPLE ──────────────────────────────────────────────────────
import json

S-001,Widget A,3,29.99,North
S-002,Widget B,1,49.99,South
S-003,Widget A,5,29.99,East
S-004,Widget C,2,89.99,North
S-005,Widget B,4,49.99,West
"""

def extract_sales_to_kafka(csv_text: str, broker: KafkaBroker) -> int:
    """Extract sales from CSV and produce each row as a Kafka event."""
    producer = KafkaProducer(broker)
    reader = csv.DictReader(io.StringIO(csv_text))
    count = 0
    for row in reader:
        event = {
            "schema_version": 1,
            "sale_id":    row["sale_id"],
            "product":    row["product"],
            "qty":        int(row["qty"]),
            "unit_price": float(row["unit_price"]),
            "region":     row["region"],
            "revenue":    round(int(row["qty"]) * float(row["unit_price"]), 2),
        }
        producer.produce("sales", key=row["sale_id"], value=json.dumps(event))
        count += 1
    producer.flush()
    return count

# ══════════════════════════════════════════════════════════════
#  CONCEPT 3 — CONSUMING EVENTS: OFFSETS AND CONSUMER GROUPS
# ══════════════════════════════════════════════════════════════
#
#  Offset management
#  ─────────────────
#  Each consumer group tracks its own offset per topic.
#  auto.offset.reset = "earliest"  start from the beginning
#  auto.offset.reset = "latest"    only new events (default)
#
#  enable.auto.commit = True    Kafka commits offset automatically
#                               after poll() returns the message.
#                               Risk: message processed but crash
#                               before next commit = re-processed.
#  enable.auto.commit = False   You commit manually after processing.
#                               Guarantees at-least-once delivery.
#
#  Consumer groups
#  ───────────────
#  Multiple consumers with the SAME group_id share a topic:
#    - partition 0 → consumer A
#    - partition 1 → consumer B
#  Multiple consumers with DIFFERENT group_ids each read ALL events:
#    - "etl-consumer" reads everything
#    - "audit-consumer" also reads everything independently
#
#  Dead letter topic
#  ─────────────────
#  If processing fails (bad JSON, schema mismatch), produce the
#  raw message to a "orders.dlq" (dead letter queue) topic.
#  Ops team inspects and replays manually.
#
# EXAMPLE ──────────────────────────────────────────────────────
import json

def transform_and_load(msg: dict, db: list, dlq: list):
    """Process one event: transform, validate, load or send to DLQ."""
    try:
        event = json.loads(msg["value"])
        if event.get("schema_version") != 1:
            raise ValueError(f"Unknown schema version: {event.get('schema_version')}")
        # Transform: add derived field
        event["revenue_band"] = (
            "high" if event["revenue"] >= 100 else
            "mid"  if event["revenue"] >= 50  else "low"
        )
        db.append(event)
    except Exception as e:
        dlq.append({"raw": msg["value"], "error": str(e)})

# ══════════════════════════════════════════════════════════════
#  CONCEPT 4 — FULL STREAM PIPELINE: EXTRACT → TOPIC → TRANSFORM → LOAD
# ══════════════════════════════════════════════════════════════
#
#  This is the pattern you used at Naya with Spark + Kafka:
#
#    Source (CSV / API / DB)
#        │
#        ▼
#    KafkaProducer  → topic "raw_events"
#        │
#        ▼
#    KafkaConsumer  (ETL group)
#        │  transform + validate
#        ▼
#    SQLite / PostgreSQL (Silver layer)
#        │
#        ▼
#    KafkaProducer  → topic "processed_events"  (for downstream consumers)
#
#  Key engineering decisions
#  ─────────────────────────
#  Batch vs stream: Kafka consumers can process one event at a time
#  (true streaming) or accumulate a micro-batch (e.g. every 100 events)
#  before writing to the database (reduces write amplification).
#
#  Idempotency: if a consumer crashes after loading but before
#  committing offset, it re-processes on restart. Use INSERT OR IGNORE
#  or ON CONFLICT DO NOTHING with the event key as the primary key.
#
#  Back-pressure: if the consumer is slow, events pile up in the topic.
#  Monitor consumer lag = latest_offset - consumer_offset.
#
# EXAMPLE ──────────────────────────────────────────────────────
import json
import sqlite3

# ── Full pipeline in simulation ───────────────────────────────

def run_pipeline(raw_data: list[dict]) -> dict:
    """
    Stage 1: Produce raw events to Kafka.
    Stage 2: Consume, transform, load to SQLite (idempotent).
    Stage 3: Produce processed events for downstream consumers.
    Returns pipeline run stats.
    """
    broker = KafkaBroker()
    db = sqlite3.connect(":memory:")
    db.execute("""
        CREATE TABLE sales (
            sale_id    TEXT PRIMARY KEY,
            product    TEXT,
            revenue    REAL,
            region     TEXT,
            band       TEXT
        )
    """)

    # Stage 1 — Extract → topic
    producer = KafkaProducer(broker)
    for row in raw_data:
        producer.produce("raw_sales", key=row["sale_id"], value=json.dumps(row))
    producer.flush()
    print(f"Stage 1: produced {len(raw_data)} events → 'raw_sales'")

    # Stage 2 — Consume → transform → load
    consumer = KafkaConsumer(broker, group_id="etl")
    consumer.subscribe(["raw_sales"])
    loaded, skipped = 0, 0
    processed_producer = KafkaProducer(broker)

    while True:
        msg = consumer.poll()
        if msg is None:
            break
        try:
            ev = json.loads(msg["value"])
            band = "high" if ev["revenue"] >= 100 else "mid" if ev["revenue"] >= 50 else "low"
            db.execute(
                "INSERT OR IGNORE INTO sales VALUES (?,?,?,?,?)",
                (ev["sale_id"], ev["product"], ev["revenue"], ev["region"], band)
            )
            processed_producer.produce(
                "processed_sales", key=ev["sale_id"],
                value=json.dumps({**ev, "band": band})
            )
            loaded += 1
        except Exception as e:
            skipped += 1
    db.commit()
    processed_producer.flush()
    print(f"Stage 2: loaded={loaded}  skipped={skipped}")

    # Stage 3 — Show downstream consumer
    downstream = KafkaConsumer(broker, group_id="dashboard")
    downstream.subscribe(["processed_sales"])
    rows = []
    while True:
        msg = downstream.poll()
        if msg is None: break
        rows.append(json.loads(msg["value"]))
    print(f"Stage 3: downstream consumer received {len(rows)} processed events")

    # Verify DB
    cursor = db.execute("SELECT band, COUNT(*), SUM(revenue) FROM sales GROUP BY band")
    print("\nDB summary:")
    for band, cnt, rev in cursor:
        print(f"  {band:4s}  count={cnt}  total_revenue={rev:.2f}")

    return {"produced": len(raw_data), "loaded": loaded, "skipped": skipped}
