Concept 1 — IR Lifecycle: PICERL

The PICERL framework (Preparation, Identification, Containment, Eradication, Recovery, Lessons Learned) gives every IR engagement a shared vocabulary and sequencing contract. Without it, IR teams skip steps under pressure — typically skipping Eradication and jumping to Recovery, which leaves the attacker's persistence mechanism intact.

Analogy to data engineering: PICERL maps to the incident lifecycle in an on-call data platform. Preparation = runbooks + dashboards. Identification = alerting. Containment = circuit breaker / rate limiter. Eradication = root cause fix. Recovery = canary redeploy. Lessons Learned = post-mortem + backlog item.

PICERL phase responsibilities:

Phase Goal Key outputs SLA target
Preparation Pre-build detection, playbooks, access Runbook library, contact list Ongoing
Identification Confirm a security incident occurred Incident ticket, initial severity < 15 min from alert
Containment Stop spread, preserve evidence Isolation decision, forensic image < 1 hr (P1)
Eradication Remove attacker tools and access Clean bill of health per system < 4 hr (P1)
Recovery Restore normal operations safely Production restored, monitoring elevated < 8 hr (P1)
Lessons Learned Prevent recurrence Post-mortem doc, detection gap tickets < 5 business days 
from dataclasses import dataclass, field
from datetime import datetime
from enum import Enum
from typing import Optional

class IRPhase(Enum):
    PREPARATION = "preparation"
    IDENTIFICATION = "identification"
    CONTAINMENT = "containment"
    ERADICATION = "eradication"
    RECOVERY = "recovery"
    LESSONS_LEARNED = "lessons_learned"

class Severity(Enum):
    P1 = "P1"   # Critical — active breach, data exfiltration
    P2 = "P2"   # High — confirmed compromise, no active exfil
    P3 = "P3"   # Medium — suspected compromise, investigation
    P4 = "P4"   # Low — precautionary / informational

@dataclass
class IRPhaseRecord:
    phase: IRPhase
    start_time: Optional[datetime]
    end_time: Optional[datetime]
    owner: str
    actions_taken: list[str] = field(default_factory=list)
    artifacts: list[str] = field(default_factory=list)

    def duration_minutes(self) -> Optional[float]:
        if self.start_time and self.end_time:
            return (self.end_time - self.start_time).total_seconds() / 60.0
        return None

@dataclass
class IncidentRecord:
    incident_id: str
    title: str
    severity: Severity
    created_at: datetime
    phase_records: dict[IRPhase, IRPhaseRecord] = field(default_factory=dict)
    affected_systems: list[str] = field(default_factory=list)
    iocs: list[str] = field(default_factory=list)
    closed_at: Optional[datetime] = None

    def current_phase(self) -> Optional[IRPhase]:
        completed = {p for p, r in self.phase_records.items() if r.end_time is not None}
        phase_order = list(IRPhase)
        for phase in reversed(phase_order):
            if phase in completed:
                idx = phase_order.index(phase)
                next_idx = idx + 1
                if next_idx < len(phase_order):
                    return phase_order[next_idx]
        return IRPhase.PREPARATION

    def total_duration_minutes(self) -> Optional[float]:
        if self.closed_at:
            return (self.closed_at - self.created_at).total_seconds() / 60.0
        return None

    def mttr_minutes(self) -> Optional[float]:
        recovery = self.phase_records.get(IRPhase.RECOVERY)
        if recovery and recovery.end_time:
            return (recovery.end_time - self.created_at).total_seconds() / 60.0
        return None

Evidence collection in IR follows the same ordering constraint as forensic analysis: volatile evidence first (RAM, running processes, network connections), then semi-volatile (event logs, prefetch), then non-volatile (disk image). class VolatilityLevel(IntEnum):
    CRITICAL = 1    # disappears on reboot
    HIGH = 2        # changes frequently
    MEDIUM = 3      # relatively stable
    LOW = 4         # persists across reboots

@dataclass
class EvidenceItem:
    name: str
    description: str
    volatility: VolatilityLevel
    collection_command: str
    output_path: str
    collected_at: Optional[datetime] = None
    sha256: Optional[str] = None
    size_bytes: Optional[int] = None

    def record_collection(self, content: bytes) -> None:
        self.collected_at = datetime.utcnow()
        self.sha256 = hashlib.sha256(content).hexdigest()
        self.size_bytes = len(content)

    def chain_of_custody_entry(self) -> dict:
        return {
            "evidence_name": self.name,
            "output_path": self.output_path,
            "collected_at": self.collected_at.isoformat() if self.collected_at else None,
            "sha256": self.sha256,
            "size_bytes": self.size_bytes,
            "volatility": self.volatility.name,
        }

from dataclasses import dataclass
from enum import Enum

class DataExfiltrationStatus(Enum):
    CONFIRMED = "confirmed"
    SUSPECTED = "suspected"
    UNLIKELY = "unlikely"
    NO = "no"

class PersistenceStatus(Enum):
    CONFIRMED = "confirmed"
    LIKELY = "likely"
    UNKNOWN = "unknown"
    NO = "no"

@dataclass
class SeverityInput:
    exfiltration: DataExfiltrationStatus
    affected_system_count: int
    crown_jewel_affected: bool
    persistence: PersistenceStatus
    c2_active: bool
    business_impact_score: int   # 0-3: 0=none, 1=minor, 2=operational, 3=revenue/legal

def classify_severity(inputs: SeverityInput) -> Severity:
    if (inputs.exfiltration == DataExfiltrationStatus.CONFIRMED or
        inputs.c2_active or
        (inputs.crown_jewel_affected and inputs.persistence == PersistenceStatus.CONFIRMED) or
        inputs.business_impact_score == 3):
        return Severity.P1

    if (inputs.exfiltration == DataExfiltrationStatus.SUSPECTED or
        inputs.affected_system_count >= 5 or
        inputs.persistence == PersistenceStatus.CONFIRMED or
        inputs.business_impact_score == 2):
        return Severity.P2

    if (inputs.persistence == PersistenceStatus.LIKELY or
        inputs.affected_system_count >= 2 or
        inputs.business_impact_score == 1):
        return Severity.P3

    return Severity.P4

Concept 1 — Building an Incident Timeline


An incident timeline is the primary deliverable of IR analysis. It sequences all attacker and defender actions with timestamps and source evidence. Without it, the team cannot identify the initial compromise vector, measure dwell time, or scope the investigation.


Timeline entry schema:


from dataclasses import dataclass, field
from datetime import datetime
from enum import Enum
from typing import Optional

class TimelineEventType(Enum):
    ATTACKER_ACTION = "attacker_action"
    DEFENDER_ACTION = "defender_action"
    SYSTEM_EVENT = "system_event"
    DETECTION = "detection"
    CONTAINMENT = "containment"

@dataclass
class TimelineEntry:
    timestamp: datetime
    event_type: TimelineEventType
    source: str             # log source: "Sysmon", "SigninLogs", "Zeek", "Analyst"
    source_event_id: str    # e.g. "EventID:4624" or "alert:INC-2024-001-01"
    host: str
    actor: str              # user, process, or IP
    action: str             # what happened
    technique_id: Optional[str] = None
    evidence_path: Optional[str] = None
    confidence: str = "high"   # high / medium / low

    def to_dict(self) -> dict:
        return {
            "timestamp": self.timestamp.isoformat(),
            "event_type": self.event_type.value,
            "source": self.source,
            "source_event_id": self.source_event_id,
            "host": self.host,
            "actor": self.actor,
            "action": self.action,
            "technique_id": self.technique_id,
            "evidence_path": self.evidence_path,
            "confidence": self.confidence,
        }

@dataclass
class IncidentTimeline:
    incident_id: str
    entries: list[TimelineEntry] = field(default_factory=list)

    def add(self, entry: TimelineEntry) -> None:
        self.entries.append(entry)
        self.entries.sort(key=lambda e: e.timestamp)

    def attacker_dwell_time_minutes(self) -> Optional[float]:
        attacker_entries = [e for e in self.entries if e.event_type == TimelineEventType.ATTACKER_ACTION]
        containment_entries = [e for e in self.entries if e.event_type == TimelineEventType.CONTAINMENT]
        if attacker_entries and containment_entries:
            first_attacker = attacker_entries[0].timestamp
            first_containment = containment_entries[0].timestamp
            return (first_containment - first_attacker).total_seconds() / 60.0
        return None

    def unique_techniques(self) -> list[str]:
        return sorted({e.technique_id for e in self.entries if e.technique_id})

    def export_json(self) -> dict:
        return {
            "incident_id": self.incident_id,
            "entry_count": len(self.entries),
            "dwell_time_minutes": self.attacker_dwell_time_minutes(),
            "techniques_observed": self.unique_techniques(),
            "entries": [e.to_dict() for e in self.entries],
        }

Concept 2 — Multi-Source Log Correlation


Real incidents span multiple log sources simultaneously. Correlating them requires a shared key — usually a timestamp window, a username, or a hostname — and a merge strategy that handles clock skew between sources (different systems may report the same event with offsets of up to 5 seconds).


import pandas as pd
from typing import Iterator

CLOCK_SKEW_TOLERANCE_SECONDS = 5

def normalize_log_timestamp(ts: str, source: str) -> datetime:
    ts = ts.replace("Z", "+00:00")
    try:
        return datetime.fromisoformat(ts).replace(tzinfo=None)
    except ValueError:
        return datetime.strptime(ts, "%Y-%m-%d %H:%M:%S")

def build_timeline_from_logs(
    signin_logs: list[dict],
    audit_logs: list[dict],
    sysmon_logs: list[dict],
    incident_id: str,
) -> IncidentTimeline:
    timeline = IncidentTimeline(incident_id=incident_id)

    for log in signin_logs:
        if log.get("ResultType") == "0":
            timeline.add(TimelineEntry(
                timestamp=normalize_log_timestamp(log["TimeGenerated"], "SigninLogs"),
                event_type=TimelineEventType.ATTACKER_ACTION,
                source="SigninLogs",
                source_event_id=log.get("CorrelationId", ""),
                host="AzureAD",
                actor=f"{log['UserPrincipalName']} from {log['IPAddress']}",
                action=f"Successful sign-in from {log.get('Location', 'Unknown')}",
                technique_id="T1078",
                confidence="high",
            ))

    for log in audit_logs:
        operation = log.get("OperationName", "")
        if "credential" in operation.lower() or "policy" in operation.lower():
            timeline.add(TimelineEntry(
                timestamp=normalize_log_timestamp(log["TimeGenerated"], "AuditLogs"),
                event_type=TimelineEventType.ATTACKER_ACTION,
                source="AuditLogs",
                source_event_id=log.get("Id", ""),
                host="AzureAD",
                actor=log.get("InitiatedBy", {}).get("user", {}).get("userPrincipalName", ""),
                action=f"AD change: {operation}",
                technique_id="T1098",
                confidence="high",
            ))

    for log in sysmon_logs:
        event_id = log.get("EventID")
        if event_id in (1, 3, 10, 25):
            eid_map = {1: "ProcessCreate", 3: "NetworkConnect", 10: "ProcessAccess", 25: "ProcessTampering"}
            technique_map = {1: "T1059", 3: "T1071", 10: "T1055", 25: "T1055.012"}
            timeline.add(TimelineEntry(
                timestamp=normalize_log_timestamp(log["UtcTime"], "Sysmon"),
                event_type=TimelineEventType.SYSTEM_EVENT,
                source="Sysmon",
                source_event_id=f"Sysmon:{event_id}:{log.get('ProcessId', '')}",
                host=log.get("Computer", ""),
                actor=log.get("Image", ""),
                action=f"Sysmon {eid_map.get(event_id, event_id)}: {log.get('CommandLine', log.get('DestinationIp', ''))}",
                technique_id=technique_map.get(event_id),
                confidence="high",
            ))

    return timeline

Concept 3 — Containment Strategy Selection


Containment is a risk vs. speed tradeoff. Aggressive containment (pull network cable, disable account immediately) stops the attacker but may destroy volatile evidence and tip off the attacker (if they have monitoring). Graduated containment (monitor, then isolate, then block) preserves more evidence but allows more damage.


from enum import Enum
from dataclasses import dataclass

class ContainmentApproach(Enum):
    AGGRESSIVE = "aggressive"      # isolate immediately — accept evidence loss
    GRADUATED = "graduated"        # monitor first, then contain in stages
    DECEPTION = "deception"        # let attacker proceed in honeynet — maximum intelligence

from dataclasses import dataclass, field
from datetime import datetime
from typing import Callable, Optional

@dataclass
class TriageStepResult:
    step_name: str
    success: bool
    output: dict
    error: Optional[str] = None
    duration_seconds: float = 0.0

@dataclass
class TriageResult:
    incident_id: str
    alert_id: str
    started_at: datetime
    completed_at: Optional[datetime] = None
    steps: list[TriageStepResult] = field(default_factory=list)
    severity: Optional[Severity] = None
    timeline: Optional[IncidentTimeline] = None
    suppressed: bool = False
    suppression_reason: str = ""

    def succeeded(self) -> bool:
        return all(s.success for s in self.steps if not s.error)

    def failed_steps(self) -> list[str]:
        return [s.step_name for s in self.steps if not s.success]

    def duration_seconds(self) -> Optional[float]:
        if self.completed_at:
            return (self.completed_at - self.started_at).total_seconds()
        return None

TriageStep = Callable[[dict, TriageResult], TriageStepResult]

def run_triage_pipeline(alert: dict, steps: list[TriageStep]) -> TriageResult:
    result = TriageResult(
        incident_id=f"INC-{datetime.utcnow().strftime('%Y%m%d-%H%M%S')}",
        alert_id=alert.get("alert_id", "unknown"),
        started_at=datetime.utcnow(),
    )
    for step_fn in steps:
        import time
        start = time.monotonic()
        try:
            step_result = step_fn(alert, result)
        except Exception as e:
            step_result = TriageStepResult(
                step_name=step_fn.__name__,
                success=False,
                output={},
                error=str(e),
                duration_seconds=time.monotonic() - start,
            )
        step_result.duration_seconds = time.monotonic() - start
        result.steps.append(step_result)
        if not step_result.success and step_result.error:
            break
    result.completed_at = datetime.utcnow()
    return result

Concept 2 — Playbook as Code


A playbook-as-code approach stores IR procedures as Python dataclasses with preconditions, actions, and expected outputs. This enables dry-run testing, version control, and automatic execution when all preconditions are met.


from dataclasses import dataclass, field
from typing import Callable, Optional
from enum import Enum

class PlaybookStatus(Enum):
    PENDING = "pending"
    RUNNING = "running"
    COMPLETED = "completed"
    FAILED = "failed"
    SKIPPED = "skipped"

@dataclass
class PlaybookAction:
    name: str
    description: str
    requires_approval: bool
    auto_executable: bool
    execute: Optional[Callable[[dict], dict]] = None

@dataclass
class PlaybookStep:
    step_id: str
    phase: IRPhase
    actions: list[PlaybookAction]
    precondition: Optional[Callable[[TriageResult], bool]] = None
    status: PlaybookStatus = PlaybookStatus.PENDING

    def can_execute(self, triage: TriageResult) -> bool:
        if self.precondition is None:
            return True
        return self.precondition(triage)

@dataclass
class Playbook:
    name: str
    trigger_rule: str
    severity_scope: list[Severity]
    steps: list[PlaybookStep]

    def applicable_to(self, triage: TriageResult) -> bool:
        return triage.severity in self.severity_scope

    def dry_run_report(self, triage: TriageResult) -> list[dict]:
        report = []
        for step in self.steps:
            can_exec = step.can_execute(triage)
            for action in step.actions:
                report.append({
                    "step": step.step_id,
                    "phase": step.phase.value,
                    "action": action.name,
                    "auto_executable": action.auto_executable and can_exec,
                    "requires_approval": action.requires_approval,
                    "precondition_met": can_exec,
                })
        return report

Concept 3 — IOC Extraction and Distribution


IOCs (Indicators of Compromise) extracted from an incident must be normalized and distributed to all enforcement points: firewall, email gateway, endpoint EDR, SIEM watchlist, and TI platform. Manual IOC distribution is slow and inconsistent — automation ensures zero-gap propagation.


from dataclasses import dataclass, field
from enum import Enum
from datetime import datetime
from typing import Optional
import re
import ipaddress

class IOCType(Enum):
    IPV4 = "ipv4"
    DOMAIN = "domain"
    URL = "url"
    FILE_HASH_MD5 = "file_hash_md5"
    FILE_HASH_SHA256 = "file_hash_sha256"
    EMAIL = "email"
    USER_ACCOUNT = "user_account"

@dataclass
class IOC:
    value: str
    ioc_type: IOCType
    confidence: str      # high / medium / low
    source: str          # "analyst", "sysmon", "zeek", "ti_feed"
    first_seen: datetime
    last_seen: datetime
    incident_id: str
    context: str = ""
    ttl_hours: int = 24  # how long to block/alert on this IOC

    def is_expired(self, as_of: Optional[datetime] = None) -> bool:
        as_of = as_of or datetime.utcnow()
        age_hours = (as_of - self.first_seen).total_seconds() / 3600
        return age_hours > self.ttl_hours

def extract_iocs_from_timeline(timeline: IncidentTimeline) -> list[IOC]:
    iocs: list[IOC] = []
    ip_pattern = re.compile(r'\b(?:\d{1,3}\.){3}\d{1,3}\b')
    email_pattern = re.compile(r'\b[A-Za-z0-9._%+-]+@[A-Za-z0-9.-]+\.[A-Z|a-z]{2,}\b')
    hash_sha256 = re.compile(r'\b[a-fA-F0-9]{64}\b')
    hash_md5 = re.compile(r'\b[a-fA-F0-9]{32}\b')

    def is_private_ip(ip: str) -> bool:
        try:
            return ipaddress.ip_address(ip).is_private
        except ValueError:
            return False

    seen: set[str] = set()
    for entry in timeline.entries:
        text = f"{entry.actor} {entry.action}"
        for ip in ip_pattern.findall(text):
            if ip not in seen and not is_private_ip(ip):
                seen.add(ip)
                iocs.append(IOC(
                    value=ip, ioc_type=IOCType.IPV4, confidence="high",
                    source=entry.source, first_seen=entry.timestamp,
                    last_seen=entry.timestamp, incident_id=timeline.incident_id,
                    context=entry.action[:100],
                ))
        for email in email_pattern.findall(text):
            if email not in seen:
                seen.add(email)
                iocs.append(IOC(
                    value=email, ioc_type=IOCType.EMAIL, confidence="medium",
                    source=entry.source, first_seen=entry.timestamp,
                    last_seen=entry.timestamp, incident_id=timeline.incident_id,
                ))

    return iocs

Concept 1 — Post-Mortem Structure


A post-mortem (also called After Action Review or Lessons Learned) is the final phase of PICERL. Its purpose is not to assign blame but to find and close the systemic gaps that allowed the incident to happen and persist. In detection engineering, every incident is also a detection gap report.

Concept 2 — Metrics: MTTD, MTTR, Dwell Time


Measuring IR effectiveness requires consistent metric definitions. Without agreed definitions, MTTD (Mean Time to Detect) in one team's report means something different than another's.


from dataclasses import dataclass
from datetime import datetime
from typing import Optional
from statistics import mean, stdev

@dataclass
class IRMetricsRecord:
    incident_id: str
    severity: Severity
    attack_start: datetime      # first attacker action observed in timeline
    first_alert: datetime       # first SIEM alert
    analyst_acknowledged: datetime  # analyst opened the ticket
    contained_at: datetime      # attacker action stopped
    recovered_at: datetime      # systems confirmed clean and operational

    def mttd_minutes(self) -> float:
        return (self.first_alert - self.attack_start).total_seconds() / 60.0

    def mtta_minutes(self) -> float:
        return (self.analyst_acknowledged - self.first_alert).total_seconds() / 60.0

    def mttc_minutes(self) -> float:
        return (self.contained_at - self.analyst_acknowledged).total_seconds() / 60.0

    def mttr_minutes(self) -> float:
        return (self.recovered_at - self.attack_start).total_seconds() / 60.0

    def dwell_time_minutes(self) -> float:
        return (self.contained_at - self.attack_start).total_seconds() / 60.0

Concept 3 — Lessons Learned to Detection Backlog


The most valuable output of a post-mortem is a concrete set of detection engineering backlog items. Every gap identified must become a ticket with a technique ID, a data source requirement, an estimated effort, and an owner.


from dataclasses import dataclass, field
from datetime import datetime, date
from typing import Optional

@dataclass
class DetectionBacklogItem:
    item_id: str
    source_incident: str
    technique_id: str
    technique_name: str
    gap_description: str
    proposed_rule_title: str
    data_source_required: str
    data_source_available: bool
    estimated_effort_days: float
    owner: Optional[str]
    target_completion: Optional[date]
    status: str = "backlog"  # backlog / in_progress / done / wontfix

    def roi_score(self) -> float:
        # Higher score = higher priority
        priority_map = {"T1078": 90, "T1098": 85, "T1059": 80, "T1055": 75}
        technique_score = priority_map.get(self.technique_id[:5], 50)
        effort_penalty = min(self.estimated_effort_days * 5, 40)
        availability_bonus = 20 if self.data_source_available else 0
        return technique_score - effort_penalty + availability_bonus

def postmortem_to_backlog(pm: PostMortem) -> list[DetectionBacklogItem]:
    items = []
    for i, gap in enumerate(pm.detection_gaps):
        items.append(DetectionBacklogItem(
            item_id=f"DET-{pm.incident_id}-{i+1:03d}",
            source_incident=pm.incident_id,
            technique_id=gap.technique_id,
            technique_name=gap.technique_id,
            gap_description=gap.description,
            proposed_rule_title=f"Detect {gap.technique_id} — {gap.description[:40]}",
            data_source_required=gap.root_cause,
            data_source_available=True,
            estimated_effort_days=gap.estimated_effort_days,
            owner=None,
            target_completion=None,
            status="backlog",
        ))
    return sorted(items, key=lambda x: -x.roi_score())

backlog_items = postmortem_to_backlog(pm)

from dataclasses import dataclass

@dataclass

    total_incidents: int
    by_severity: dict[str, int]
    mean_mttd: float
    mean_mttr: float
    mean_dwell: float

    open_detection_gaps: int
    detection_gap_effort_days: float

def build_ir_dashboard(
    packages: list[TriagePackage],
    postmortems: list[PostMortem],
    period_label: str = "Last 30 days",
) -> IRDashboardData:
    by_severity: dict[str, int] = {}
    all_techniques: list[str] = []

    for pkg in packages:
        sev = pkg.incident.severity.value
        by_severity[sev] = by_severity.get(sev, 0) + 1
        all_techniques.extend(pkg.timeline.unique_techniques())

    technique_counts: dict[str, int] = {}
    for t in all_techniques:
        technique_counts[t] = technique_counts.get(t, 0) + 1
    top_techniques = sorted(technique_counts.items(), key=lambda x: -x[1])[:5]

    open_gaps = sum(len(pm.detection_gaps) for pm in postmortems)
    total_effort = sum(pm.effort_estimate_total_days() for pm in postmortems)

    return IRDashboardData(
        period_label=period_label,
        total_incidents=len(packages),
        by_severity=by_severity,
        mean_mttd=0.0,
        mean_mttr=0.0,
        mean_dwell=0.0,
        top_techniques=top_techniques,
        open_detection_gaps=open_gaps,
        detection_gap_effort_days=total_effort,
    )

def render_ir_dashboard_md(data: IRDashboardData) -> str:
    lines = [
        f"# IR Dashboard — {data.period_label}",
        "",
        "## Summary",
        f"- Total incidents: {data.total_incidents}",
        f"- By severity: {data.by_severity}",
        f"- Mean MTTD: {data.mean_mttd:.1f} min",
        f"- Mean MTTR: {data.mean_mttr:.1f} min",
        f"- Mean dwell: {data.mean_dwell:.1f} min",
        "",
        "## Top Techniques",
        "| Technique | Count |",
        "|-----------|-------|",
    ]
    for tech, count in data.top_techniques:
        lines.append(f"| {tech} | {count} |")
    lines += [
        "",
        "## Detection Backlog",
        f"- Open gaps: {data.open_detection_gaps}",
        f"- Estimated effort: {data.detection_gap_effort_days:.1f} days",
    ]
    return "\n".join(lines)
